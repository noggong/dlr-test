    35	    suspend fun group(size: Int, date: LocalDate = DATE): SignalGroup {

    42	            db.sql("SELECT id, pk_hash FROM questions WHERE difficulty = :d ORDER BY id LIMIT 1")
    43	                .bind("d", no)
    44	                .map { row -> row.get("id", Number::class.java)!!.toLong() to row.get("pk_hash", String::class.java)!! }
    45	                .one().awaitSingle()
    46	        }

    48	            val uid = db.sql("INSERT INTO users (login_id, password_hash, group_id) VALUES (:l, 'x', :g) RETURNING id")
    49	                .bind("l", "ss4$tag$i").bind("g", gid)
    50	                .map { row -> row.get("id", Number::class.java)!!.toLong() }.one().awaitSingle()
    51	            db.sql(
    52	                "INSERT INTO daily_progress (game_date, user_id, group_id, progress, status, started_at, is_late) " +
    53	                    "VALUES (:d, :u, :g, 0, 'IN_PROGRESS', now(), false)",
    54	            ).bind("d", date).bind("u", uid).bind("g", gid).then().awaitFirstOrNull()

    58	                        "VALUES (:u, :q, :d, :no, 'SHOWN', now())",

    60	            }
    61	            uid

    99	    fun decided(
   100	        g: SignalGroup,
   101	        result: String,
   102	        members: List<Member>,
   103	        at: Long = nextUs(),
   104	        rankingFinal: Int = 1,
   105	    ): Map<String, String> {

   139	    /** 표 [table] 에서 이 그룹(또는 그룹원)의 행 전부 — 비교용(정렬 고정). */
   140	    suspend fun rows(table: String, g: SignalGroup): List<Map<String, Any?>> {
   141	        val where = when (table) {
   142	            "group_results", "group_result_members" -> "game_date = :d AND group_id = ${g.gid}"
   143	            "served_questions" -> "user_id IN (${g.uids.joinToString()})"
   144	            else -> "game_date = :d AND user_id IN (${g.uids.joinToString()})"
   145	        }
   146	        val order = if (table == "group_results") "group_id" else if (table == "served_questions") "user_id, question_id" else "user_id"
   147	        var spec = db.sql("SELECT * FROM $table WHERE $where ORDER BY $order")
   148	        if (":d" in where) spec = spec.bind("d", g.date)
   149	        return spec.fetch().all().collectList().awaitSingle()
   150	    }
