+/**

+ *
+ * 칸을 [SignalRecords] 로 읽고(틀리면 [MalformedEventException]) 저장소 메서드 하나(= 한 트랜잭션)를 부른다 — 돌아오면 커밋된 것이다.

+) : PersisterEventHandler {
+    override val types: Set<String> = SignalRecords.TYPES
+
+    override suspend fun handle(event: StreamEvent) {
+        val record = try {
+            SignalRecords.parse(event.fields)

+            throw MalformedEventException("${event.type} ${event.id}: ${e.message}", e)
+        }
+        when (record) {
+            is SolveRecord -> repository.recordSolve(record)
+            is PassRecord -> repository.recordPass(record)
+            is CheatFailRecord -> repository.recordCheatFail(record)
+            is GroupDecidedRecord -> repository.recordGroupDecided(record)
+        }
+    }

+/** Stream 칸(값은 모두 문자열 — HLD platform §4.2) → [SignalRecord]. 칸이 없거나 형식이 틀리면 [IllegalArgumentException]. */
+object SignalRecords {
+    const val SOLVE_APPLIED = "solve.applied"
+    const val PASS_USED = "pass.used"
+    const val MEMBER_CHEAT_FAILED = "member.cheat_failed"
+    const val GROUP_DECIDED = "group.decided"
+
+    val TYPES: Set<String> = setOf(SOLVE_APPLIED, PASS_USED, MEMBER_CHEAT_FAILED, GROUP_DECIDED)
+
+    private val json = JsonMapper.builder().build()
+    private val PK = Regex("[0-9a-f]{64}")
+
+    fun parse(fields: Map<String, String>): SignalRecord {
+        val f = Fields(fields)
+        return when (val type = f.req("type")) {
+            SOLVE_APPLIED -> SolveRecord(
+                date = f.date(),
+                userId = f.uid(),
+                questionNo = f.questionNo(),
+                questionPk = f.pk(),
+                successTime = f.iso("successTime"),
+                clickedAt = f.micros("clickedAt"),
+                arrivedAt = f.micros("arrivedAt"),
+            )
+            PASS_USED -> PassRecord(f.date(), f.uid(), f.questionNo(), f.pk(), f.micros("at"))
+            MEMBER_CHEAT_FAILED -> CheatFailRecord(f.date(), f.uid(), f.micros("cheatFailedAt"))
+            GROUP_DECIDED -> {
+                val memberCount = f.int("memberCount")
+                require(memberCount >= 1) { "memberCount 는 1 이상: $memberCount" }
+                val members = ranking(f.req("ranking"))
+                require(members.size == memberCount) { "ranking 줄 수 ${members.size} ≠ memberCount $memberCount" }
+                GroupDecidedRecord(
+                    date = f.date(),
+                    groupId = GroupId(f.long("gid")),
+                    result = enumValue("result", f.req("result")),
+                    decidedAt = f.micros("decidedAt"),
+                    memberCount = memberCount,
+                    members = members,
+                )
+            }
+            else -> throw IllegalArgumentException("이 기능의 type 이 아님: $type")
+        }
+    }
+
+    private fun ranking(raw: String): List<ResultMemberRecord> {
+        val rows = try {
+            json.readTree(raw)
+        } catch (e: JacksonException) {
+            throw IllegalArgumentException("ranking JSON 을 읽지 못함", e)
+        }
+        require(rows.isArray) { "ranking 이 JSON 배열이 아님" }
+        val members = ArrayList<ResultMemberRecord>()
+        for (row in rows) {
+            val progress = intOf(row, "p")
+            require(progress in -1..3) { "ranking 줄의 p 범위 밖: $row" }
+            val rank = intOf(row, "r")
+            require(rank >= 1) { "ranking 줄의 r 은 1 이상: $row" }
+            val pass = if (isAbsent(row, "n")) null else intOf(row, "n")
+            require(pass == null || pass in 1..3) { "ranking 줄의 n 범위 밖: $row" }
+            members += ResultMemberRecord(
+                userId = UserId(idOf(row)),
+                rank = rank,
+                progress = progress,
+                status = enumValue("ranking.s", textOf(row, "s")),
+                lastSolvedAt = if (isAbsent(row, "l")) null else microsToInstant(long("ranking.l", textOf(row, "l"))),
+                passNo = pass,
+            )
+        }
+        require(members.map { it.userId }.toSet().size == members.size) { "ranking 에 같은 id 가 두 줄" }
+        return members
+    }
+
+    private fun isAbsent(row: JsonNode, name: String): Boolean = row.get(name)?.isNull ?: true
+
+    private fun idOf(row: JsonNode): Long {
+        val id = row.get("id")
+        require(id != null && (id.isString || id.isIntegralNumber)) { "ranking 줄에 id 없음: $row" }
+        return if (id.isIntegralNumber) id.longValue() else long("ranking.id", id.stringValue())
+    }
+
+    private fun intOf(row: JsonNode, name: String): Int {
+        val v = row.get(name)
+        require(v != null && v.isIntegralNumber) { "ranking 줄의 $name 이 정수가 아님: $row" }
+        return v.intValue()
+    }
+
+    private fun textOf(row: JsonNode, name: String): String {
+        val v = row.get(name)
+        require(v != null && v.isString) { "ranking 줄의 $name 이 문자열이 아님: $row" }
+        return v.stringValue()
+    }
+
+    private fun long(name: String, value: String): Long =
+        requireNotNull(value.toLongOrNull()) { "$name 은 정수: $value" }
+
+    private inline fun <reified E : Enum<E>> enumValue(name: String, value: String): E =
+        requireNotNull(enumValues<E>().firstOrNull { it.name == value }) { "$name 값 틀림: $value" }
+
+    private class Fields(private val fields: Map<String, String>) {
+        fun req(name: String): String = requireNotNull(fields[name]) { "$name 칸 없음" }
+
+        fun long(name: String): Long = long(name, req(name))
+
+        fun int(name: String): Int = requireNotNull(req(name).toIntOrNull()) { "$name 은 정수: ${fields[name]}" }
+
+        fun micros(name: String): Instant = microsToInstant(long(name))
+
+        fun date(): LocalDate = try {
+            LocalDate.parse(req("date"))
+        } catch (e: DateTimeParseException) {
+            throw IllegalArgumentException("date 는 YYYY-MM-DD: ${fields["date"]}", e)
+        }
+
+        fun uid(): UserId = UserId(long("uid"))
+
+        fun questionNo(): Int = int("questionNo").also { require(it in 1..3) { "questionNo 는 1~3: $it" } }
+
+        fun pk(): String = req("questionId").also { require(PK.matches(it)) { "questionId 는 hex 64: $it" } }
+
+        fun iso(name: String): Instant = try {
+            OffsetDateTime.parse(req(name)).toInstant()
+        } catch (e: DateTimeParseException) {
+            throw IllegalArgumentException("$name 은 오프셋 있는 ISO-8601: ${fields[name]}", e)
+        }
+    }
+}

+package com.delightroom.mission.infra.rdb
+
+import com.delightroom.mission.core.domain.board.GroupResultCode
+import com.delightroom.mission.core.domain.progress.CheatFailRecord
+import com.delightroom.mission.core.domain.progress.GroupDecidedRecord

+import com.delightroom.mission.core.domain.progress.SolveRecord
+import com.delightroom.mission.core.port.SignalEventRepository
+import org.springframework.r2dbc.core.DatabaseClient
+import org.springframework.r2dbc.core.awaitOneOrNull
+import org.springframework.r2dbc.core.awaitRowsUpdated
+import org.springframework.transaction.reactive.TransactionalOperator
+import org.springframework.transaction.reactive.executeAndAwait
+import java.time.Instant
+import java.time.OffsetDateTime
+import java.time.ZoneOffset
+
+/**
+ * solve-signal 이벤트 → PostgreSQL(LLD solve-signal §2.3 · LLD platform §3). 메서드 하나 = 한 트랜잭션.
+ *
+ * 모두 멱등이다 — PK + `ON CONFLICT` · `GREATEST` · `WHERE status = 'IN_PROGRESS'` · `pass_question_no IS NULL`.
+ * 같은 이벤트를 두 번 넘겨도 두 번째는 행을 바꾸지 않는다(`updated_at` 도 조건이 맞을 때만 바뀐다).
+ */
+internal class RdbSignalEventRepository(
+    private val db: DatabaseClient,
+    private val tx: TransactionalOperator,
+) : SignalEventRepository {
+    override suspend fun recordSolve(record: SolveRecord) {
+        tx.executeAndAwait {
+            val questionId = questionId(record.questionPk)
+            db.sql(
+                "INSERT INTO solve_signals (game_date, user_id, question_no, question_id, success_time, click_time, arrived_at) " +
+                    "VALUES (:d, :u, :no, :q, :success, :click, :arrived) ON CONFLICT DO NOTHING",
+            )
+                .bind("d", record.date).bind("u", record.userId.value).bind("no", record.questionNo).bind("q", questionId)
+                .bind("success", odt(record.successTime)).bind("click", odt(record.clickedAt))
+                .bind("arrived", odt(record.arrivedAt))
+                .fetch().awaitRowsUpdated()
+            // 진행도가 이미 :no 이상이면(재처리) 아무 칸도 바뀌지 않게 WHERE 로 거른다
+            db.sql(
+                "UPDATE daily_progress SET progress = :no, last_solved_at = :arrived, " +
+                    "status = CASE WHEN status = 'IN_PROGRESS' AND :no = 3 THEN 'SUCCEEDED' ELSE status END, " +
+                    "updated_at = now() " +
+                    "WHERE game_date = :d AND user_id = :u AND progress < :no",
+            )
+                .bind("no", record.questionNo).bind("arrived", odt(record.arrivedAt))
+                .bind("d", record.date).bind("u", record.userId.value)
+                .fetch().awaitRowsUpdated()
+            db.sql("UPDATE served_questions SET kind = 'SOLVED' WHERE user_id = :u AND question_id = :q AND kind <> 'SOLVED'")
+                .bind("u", record.userId.value).bind("q", questionId)
+                .fetch().awaitRowsUpdated()
+        }
+    }
+
+    override suspend fun recordPass(record: PassRecord) {
+        tx.executeAndAwait {
+            val questionId = questionId(record.questionPk)
+            db.sql(
+                "INSERT INTO served_questions (user_id, question_id, game_date, question_no, kind, served_at) " +
+                    "VALUES (:u, :q, :d, :no, 'PASSED', :at) " +
+                    "ON CONFLICT (user_id, question_id) DO UPDATE SET kind = 'PASSED' WHERE served_questions.kind <> 'PASSED'",
+            )
+                .bind("u", record.userId.value).bind("q", questionId).bind("d", record.date)
+                .bind("no", record.questionNo).bind("at", odt(record.at))
+                .fetch().awaitRowsUpdated()
+            db.sql(
+                "UPDATE daily_progress SET pass_question_no = :no, passed_question_id = :q, updated_at = now() " +
+                    "WHERE game_date = :d AND user_id = :u AND pass_question_no IS NULL",
+            )
+                .bind("no", record.questionNo).bind("q", questionId)
+                .bind("d", record.date).bind("u", record.userId.value)
+                .fetch().awaitRowsUpdated()
+        }
+    }
+
+    override suspend fun recordCheatFail(record: CheatFailRecord) {
+        tx.executeAndAwait {
+            db.sql(
+                "UPDATE daily_progress SET status = 'CHEAT_FAILED', cheat_failed_at = :at, updated_at = now() " +
+                    "WHERE game_date = :d AND user_id = :u AND status = 'IN_PROGRESS'",
+            )
+                .bind("at", odt(record.at)).bind("d", record.date).bind("u", record.userId.value)
+                .fetch().awaitRowsUpdated()
+        }
+    }
+
+    override suspend fun recordGroupDecided(record: GroupDecidedRecord): Boolean =
+        tx.executeAndAwait {
+            val inserted = db.sql(
+                "INSERT INTO group_results (game_date, group_id, result, completed_at, member_count) " +
+                    "VALUES (:d, :g, :result, :at, :count) ON CONFLICT DO NOTHING",
+            )
+                .bind("d", record.date).bind("g", record.groupId.value).bind("result", record.result.name)
+                .bind("at", odt(record.decidedAt)).bind("count", record.memberCount)
+                .fetch().awaitRowsUpdated() == 1L
+            // 결과는 정확히 한 번(NFR-2) — 스냅샷도 결과를 처음 넣은 트랜잭션에서만 넣는다
+            if (inserted) record.members.forEach { insertMember(record, it) }
+            // 마감 실패 그룹원(LLD platform §3) — Lua deadline 이 그룹 Hash 에서 바꾼 상태를 정본에도 남긴다. 조건부라 멱등
+            if (record.result == GroupResultCode.FAILED_DEADLINE) {
+                db.sql(
+                    "UPDATE daily_progress SET status = 'FAILED', updated_at = now() " +
+                        "WHERE game_date = :d AND group_id = :g AND status IN ('NOT_STARTED', 'IN_PROGRESS')",
+                )
+                    .bind("d", record.date).bind("g", record.groupId.value)
+                    .fetch().awaitRowsUpdated()
+            }
+            inserted
+        }
+
+    private suspend fun insertMember(record: GroupDecidedRecord, member: ResultMemberRecord) {
+        var spec = db.sql(
+            "INSERT INTO group_result_members " +
+                "(game_date, group_id, user_id, rank, progress, status, started, last_solved_at, pass_question_no, cheat_failed) " +
+                "VALUES (:d, :g, :u, :rank, :p, :s, :started, :ls, :pass, :cheat) ON CONFLICT (game_date, user_id) DO NOTHING",
+        )
+            .bind("d", record.date).bind("g", record.groupId.value).bind("u", member.userId.value)
+            .bind("rank", member.rank).bind("p", member.progress).bind("s", member.status.name)
+            .bind("started", member.started).bind("cheat", member.cheatFailed)
+        spec = member.lastSolvedAt?.let { spec.bind("ls", odt(it)) } ?: spec.bindNull("ls", OffsetDateTime::class.java)
+        spec = member.passNo?.let { spec.bind("pass", it) } ?: spec.bindNull("pass", Integer::class.java)
+        spec.fetch().awaitRowsUpdated()
+    }
+
+    /**
+     * 문제 PK(hex) → `questions.id`. 없으면 예외 — 출제된 문제가 표에 없는 손상이라 기록을 버리지 않고 재처리한다
+     * (버킷이 멈춰 `persister.stuck` 경보 — LLD solve-signal §2.3).
+     */
+    private suspend fun questionId(pk: String): Long =
+        db.sql("SELECT id FROM questions WHERE pk_hash = :pk")
+            .bind("pk", pk)
+            .map { row -> row.get("id", Number::class.java)!!.toLong() }
+            .awaitOneOrNull()
+            ?: throw IllegalStateException("questions 에 없는 문제 PK: $pk")
+
+    private fun odt(instant: Instant): OffsetDateTime = instant.atOffset(ZoneOffset.UTC)
+}
