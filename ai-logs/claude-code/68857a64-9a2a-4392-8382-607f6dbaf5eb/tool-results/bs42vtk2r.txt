   166	+ * `worker: persister` 소비 루프(LLD platform §3.1) — 버킷 Stream `ev:{<b>}` 을 consumer group `persister` 로 읽어
   167	+ * `type` 별 [PersisterEventHandler] 에 넘기고, 처리기가 돌아온 것(= 커밋됨)만 ACK 한다.
   168	+ *
   169	+ * - 맡는 버킷은 `b mod instanceCount == instanceIndex`. 버킷 하나는 레인(코루틴) 하나만 돌아 Stream 순서대로 처리된다.
   170	+ * - 처리기가 실패하면 그 메시지에서 멈추고(뒤 메시지는 처리하지 않는다) 그 버킷은 backoff 뒤 자기 PEL 부터 다시 읽는다.
   171	+ * - 기동 직후와 [PersisterProperties.claimInterval] 마다 `XAUTOCLAIM` 으로 죽은 소비자의 pending 을 가져온다.
   172	+ */
   173	+@Component
   174	+class StreamPersister(
   175	+    private val stream: EventStreamConsumer,
   176	+    handlers: List<PersisterEventHandler> = emptyList(),
   177	+    private val properties: PersisterProperties,
   178	+    game: GameProperties,
   179	+) : SmartLifecycle {
   180	+    private val handlerByType: Map<String, PersisterEventHandler> = handlerMap(handlers)
   181	+
   182	+    /** 이 인스턴스가 맡는 버킷(오름차순). */
   183	+    val buckets: List<Int> =
   184	+        (0 until game.bucketCount).filter { it % properties.instanceCount == properties.instanceIndex }
   185	+
   186	+    private val consumer = properties.consumerName
   187	+
   188	+    @Volatile private var job: Job? = null
   189	+
   190	+    override fun start() {
   191	+        if (!properties.enabled || job != null) return
   192	+        val scope = CoroutineScope(SupervisorJob() + Dispatchers.Default + CoroutineName("persister"))
   193	+        val states = buckets.map(::BucketState)
   194	+        val lanes = states.withIndex().groupBy({ it.index % properties.parallelism }, { it.value })
   195	+        lanes.values.forEach { lane -> scope.launch { runLane(lane) } }
   196	+        job = scope.coroutineContext[Job]
   197	+        log.info(
   198	+            "persister.started consumer={} buckets={} lanes={} handlers={}",
   199	+            consumer, buckets.size, lanes.size, handlerByType.keys.sorted(),
   200	+        )
   201	+    }
   202	+
   203	+    override fun stop() {
   204	+        val running = job ?: return
   205	+        // 처리 중이던 메시지는 ACK 되지 않아 재기동 때 자기 PEL 로 다시 처리된다
   206	+        runBlocking { running.cancelAndJoin() }
   207	+        job = null
   208	+        log.info("persister.stopped consumer={}", consumer)
   209	+    }
   210	+

   212	+
   213	+    /** 버킷 하나의 소비 상태 — 그 버킷을 맡은 레인만 만진다. */
   214	+    private class BucketState(val bucket: Int) {

   216	+

   218	+        var backlog = true
   219	+        var claimDue = true
   220	+        var failures = 0
   221	+        var retryAtNanos = 0L
   222	+    }
   223	+
   224	+    private suspend fun runLane(lane: List<BucketState>) {
   225	+        var nextClaimAt = System.nanoTime() + properties.claimInterval.toNanos()
   226	+        while (coroutineContext.isActive) {
   227	+            if (System.nanoTime() - nextClaimAt >= 0) {
   228	+                lane.forEach { it.claimDue = true }
   229	+                nextClaimAt = System.nanoTime() + properties.claimInterval.toNanos()
   230	+            }
   231	+            var progressed = false
   232	+            for (state in lane) {
   233	+                try {
   234	+                    if (step(state)) progressed = true
   235	+                } catch (e: CancellationException) {
   236	+                    throw e
   237	+                } catch (e: Exception) {
   238	+                    // Redis 가 비었으면(재시작 · FLUSHALL) 그룹도 없다 — 다음 걸음에 다시 만든다(멱등)
   239	+                    state.groupReady = false
   240	+                    state.failures++
   241	+                    state.retryAtNanos = System.nanoTime() + backoffNanos(state.failures)
   242	+                    log.warn("persister.redis_failed bucket={} failures={} reason={}", state.bucket, state.failures, e.toString())
   243	+                }
   244	+            }
   245	+            if (!progressed) delay(properties.idleDelay.toMillis())
   246	+        }
   247	+    }
   248	+
   249	+    /** 버킷 한 걸음 — 메시지를 읽었으면 true. */
   250	+    private suspend fun step(state: BucketState): Boolean {
   251	+        if (System.nanoTime() - state.retryAtNanos < 0) return false
   252	+        val b = state.bucket
   253	+        if (!state.groupReady) {
   254	+            stream.ensureGroup(b)
   255	+            state.groupReady = true
   256	+        }
   257	+        if (state.claimDue) {
   258	+            state.claimDue = false
   259	+            var cursor = "0-0"
   260	+            var claimed = false
   261	+            do {
   262	+                val page = stream.claimIdle(b, consumer, properties.claimMinIdle, properties.batchSize, cursor)
   263	+                if (page.events.isNotEmpty()) claimed = true
   264	+                // 멈추면 가져온 것은 이미 내 PEL 이다 — backlog 가 다시 읽는다. 남은 쪽은 다음 회수 때 가져온다
   265	+                if (!process(state, page.events)) return true
   266	+                cursor = page.cursor
   267	+            } while (!page.done)
   268	+            if (claimed) return true
   269	+        }
   270	+        if (state.backlog) {
   271	+            val pending = stream.readOwnPending(b, consumer, properties.batchSize)
   272	+            if (pending.isNotEmpty()) {
   273	+                process(state, pending)
   274	+                return true

   277	+        }

   279	+        if (fresh.isEmpty()) return false
   280	+        process(state, fresh)
   281	+        return true
   282	+    }
   283	+
   284	+    /**
   285	+     * [events] 를 순서대로 처리하고 처리된 것을 ACK 한다. 처리기가 실패하면 그 메시지에서 멈추고 false —
   286	+     * 버킷은 backoff 뒤 자기 PEL 부터 다시 읽는다.
   287	+     */
   288	+    private suspend fun process(state: BucketState, events: List<StreamEvent>): Boolean {
   289	+        val done = ArrayList<String>(events.size)
   290	+        var failed: StreamEvent? = null
   291	+        try {
   292	+            for (event in events) {
   293	+                if (!handle(state, event)) {
   294	+                    failed = event
   295	+                    break
   296	+                }
   297	+                done += event.id
   298	+            }
   299	+        } finally {
   300	+            // 취소돼도 이미 커밋한 것은 ACK 한다(못 하면 다시 받아 멱등 SQL 로 같은 결과)

   302	+        }
   303	+        if (failed == null) {
   304	+            state.failures = 0
   305	+            return true
   306	+        }
   307	+        state.failures++
   308	+        state.backlog = true
   309	+        state.retryAtNanos = System.nanoTime() + backoffNanos(state.failures)
   310	+        return false
   311	+    }
   312	+
   313	+    /** 메시지 한 건 — ACK 해도 되면 true(커밋됨 · 건너뜀). 처리기가 실패하면 false. */
   314	+    private suspend fun handle(state: BucketState, event: StreamEvent): Boolean {
   315	+        val bucket = state.bucket
   316	+        if (event.deleted) {
   317	+            log.warn("persister.deleted bucket={} id={}", bucket, event.id)
   318	+            return true
   319	+        }
   320	+        val handler = handlerByType[event.type]
   321	+        if (handler == null) {
   322	+            log.warn("persister.unhandled bucket={} id={} type={}", bucket, event.id, event.type)
   323	+            return true
   324	+        }
   325	+        return try {
   326	+            handler.handle(event)
   327	+            true
   328	+        } catch (e: CancellationException) {
   329	+            throw e
   330	+        } catch (e: MalformedEventException) {
   331	+            log.error("persister.malformed bucket={} id={} type={} reason={}", bucket, event.id, event.type, e.message)
   332	+            true
   333	+        } catch (e: Exception) {
   334	+            log.warn(
   335	+                "persister.handler_failed bucket={} id={} type={} failures={} reason={}",
   336	+                bucket, event.id, event.type, state.failures + 1, e.toString(),
   337	+            )
   338	+            false
   339	+        }
   340	+    }
   341	+
   342	+    private fun backoffNanos(failures: Int): Long {
   343	+        val base = properties.retryBackoff.toNanos()
   344	+        val max = properties.retryBackoffMax.toNanos()
   345	+        val shift = (failures - 1).coerceIn(0, 30)
   346	+        return if (base > max shr shift) max else (base shl shift).coerceAtMost(max)
   347	+    }
   348	+
   349	+    private companion object {
   350	+        val log = LoggerFactory.getLogger(StreamPersister::class.java)
   351	+
   352	+        fun handlerMap(handlers: List<PersisterEventHandler>): Map<String, PersisterEventHandler> {
   353	+            val byType = HashMap<String, PersisterEventHandler>()
   354	+            for (handler in handlers) {
   355	+                for (type in handler.types) {
   356	+                    val other = byType.putIfAbsent(type, handler)
   357	+                    check(other == null) {
   358	+                        "persister 처리기가 같은 type 을 둘 맡는다: $type — ${other!!.javaClass.name} · ${handler.javaClass.name}"
   359	+                    }
   360	+                }
   361	+            }
   362	+            return byType
   363	+        }
   364	+    }

   524	+        redis.execute(ReactiveRedisCallback<String> { it.serverCommands().flushAll() }).collectList().awaitSingle()
   525	+        db.sql(
   526	+            """
   527	+            CREATE TABLE IF NOT EXISTS persister_probe (
   528	+              ord bigserial, bucket int NOT NULL, seq int NOT NULL, msg_id text NOT NULL, inst text NOT NULL,
   529	+              PRIMARY KEY (bucket, seq)
   530	+            )
   531	+            """.trimIndent(),
   532	+        ).then().awaitFirstOrNull()
   533	+        db.sql("TRUNCATE persister_probe").then().awaitFirstOrNull()
   534	+    }
   535	+
   536	+    suspend fun rows(): List<ProbeRow> =
   537	+        db.sql("SELECT bucket, seq, inst FROM persister_probe ORDER BY ord")
   538	+            .map { row ->
   539	+                ProbeRow((row.get("bucket") as Number).toInt(), (row.get("seq") as Number).toInt(), row.get("inst") as String)

   544	+    suspend fun add(bucket: Int, seq: Int, vararg extra: Pair<String, String>, type: String = ProbeHandler.PROBE): String {
   545	+        val fields = linkedMapOf("type" to type, "bucket" to "$bucket", "seq" to "$seq") + extra
   546	+        return redis.opsForStream<String, String>().add(MapRecord.create(stream(bucket), fields)).awaitSingle().value
   547	+    }
   548	+

   550	+        redis.opsForStream<String, String>().pending(stream(bucket), GROUP).awaitSingle().totalPendingMessages
   551	+
   552	+    suspend fun pendingOf(bucket: Int, consumer: String): Long =
   553	+        redis.opsForStream<String, String>().pending(stream(bucket), GROUP).awaitSingle()
   554	+            .pendingMessagesPerConsumer[consumer] ?: 0L
   555	+
   556	+    suspend fun length(bucket: Int): Long = redis.opsForStream<String, String>().size(stream(bucket)).awaitSingle()
   557	+
