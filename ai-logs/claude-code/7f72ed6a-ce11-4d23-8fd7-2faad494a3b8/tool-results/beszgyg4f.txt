    30	/**
    31	 * `worker: persister` 소비 루프(LLD platform §3.1) — 버킷 Stream `ev:{<b>}` 을 consumer group `persister` 로 읽어
    32	 * `type` 별 [PersisterEventHandler] 에 넘기고, 처리기가 돌아온 것(= 커밋됨)만 ACK 한다.
    33	 *
    34	 * - 맡는 버킷은 `b mod instanceCount == instanceIndex`. 버킷 하나는 레인(코루틴) 하나만 돌아 Stream 순서대로 처리된다.
    35	 * - 처리기가 실패하면 그 메시지에서 멈추고(뒤 메시지는 처리하지 않는다) 그 버킷은 backoff 뒤 자기 PEL 부터 다시 읽는다.
    36	 * - 같은 메시지의 센 실패가 [PersisterProperties.deadLetterAfter] 번이면(형식 오류는 처음에) [DeadLetterStore] 에 쌓고 ACK 해 다음 메시지로 간다
    37	 *   (prd D20). 일시 장애([PersisterFailures])는 세지 않고, 쌓기가 실패하면 ACK 하지 않는다.
    38	 * - 기동 직후와 [PersisterProperties.claimInterval] 마다 `XAUTOCLAIM` 으로 죽은 소비자의 pending 을 가져온다.
    39	 */
    40	@Component
    41	class StreamPersister(
    42	    private val stream: EventStreamConsumer,
    43	    handlers: List<PersisterEventHandler> = emptyList(),
    44	    private val properties: PersisterProperties,
    45	    game: GameProperties,
    46	    private val deadLetters: DeadLetterStore,
    47	    private val clock: Clock,
    48	) : SmartLifecycle {
    49	    private val handlerByType: Map<String, PersisterEventHandler> = handlerMap(handlers)
    50	
    51	    /** 이 인스턴스가 맡는 버킷(오름차순). */
    52	    val buckets: List<Int> =
    53	        (0 until game.bucketCount).filter { it % properties.instanceCount == properties.instanceIndex }
    54	
    55	    private val consumer = properties.consumerName
    56	
    57	    @Volatile private var job: Job? = null
    58	
    59	    override fun start() {
    60	        if (!properties.enabled || job != null) return
    61	        val scope = CoroutineScope(SupervisorJob() + Dispatchers.Default + CoroutineName("persister"))
    62	        val states = buckets.map(::BucketState)
    63	        val lanes = states.withIndex().groupBy({ it.index % properties.parallelism }, { it.value })
    64	        lanes.values.forEach { lane -> scope.launch { runLane(lane) } }
    65	        job = scope.coroutineContext[Job]
    66	        log.info(
    67	            "persister.started consumer={} buckets={} lanes={} handlers={}",
    68	            consumer, buckets.size, lanes.size, handlerByType.keys.sorted(),
    69	        )
    70	    }
    71	
    72	    override fun stop() {
    73	        val running = job ?: return
    74	        // 처리 중이던 메시지는 ACK 되지 않아 재기동 때 자기 PEL 로 다시 처리된다
    75	        runBlocking { running.cancelAndJoin() }
    76	        job = null
    77	        log.info("persister.stopped consumer={}", consumer)
    78	    }
    79	
    80	    override fun isRunning(): Boolean = job != null
    81	
    82	    /** 버킷 하나의 소비 상태 — 그 버킷을 맡은 레인만 만진다. */
    83	    private class BucketState(val bucket: Int) {
    84	        var groupReady = false
    85	
    86	        /** 내게 배달됐지만 ACK 하지 않은 메시지가 있을 수 있다(기동 직후 · 처리기 실패 · Redis 오류 뒤) — 자기 PEL 부터 읽는다. */
    87	        var backlog = true
    88	        var claimDue = true
    89	
    90	        /** 처리기 연속 실패 수 — 메시지를 처리하면 0. */
    91	        var failures = 0
    92	
    93	        /** 센 실패를 세는 메시지 · 센 수 · 처음 센 시각(LLD §3.1 dead-letter 2) — 다른 메시지가 실패하면 새로 센다. */
    94	        var strikeId: String? = null
    95	        var strikes = 0
    96	        var firstStrikeAt: Instant? = null
    97	
    98	        fun clearStrikes() {
    99	            strikeId = null
   100	            strikes = 0
   101	            firstStrikeAt = null
   102	        }
   103	
   104	        /** Redis 연속 오류 수 — 오류 없이 한 걸음을 마치면 0. */
   105	        var redisFailures = 0
   106	        var retryAtNanos = 0L
   107	    }
   108	
   109	    private suspend fun runLane(lane: List<BucketState>) {
   110	        var nextClaimAt = System.nanoTime() + properties.claimInterval.toNanos()
   111	        while (coroutineContext.isActive) {
   112	            if (System.nanoTime() - nextClaimAt >= 0) {
   113	                lane.forEach { it.claimDue = true }
   114	                nextClaimAt = System.nanoTime() + properties.claimInterval.toNanos()
   115	            }
   116	            var progressed = false
   117	            for (state in lane) {
   118	                if (System.nanoTime() - state.retryAtNanos < 0) continue
   119	                try {
   120	                    if (step(state)) progressed = true
   121	                    state.redisFailures = 0
   122	                } catch (e: Throwable) {
   123	                    if (e is CancellationException && !currentCoroutineContext().isActive) throw e
   124	                    // 읽기가 Redis 에서는 됐는데 응답을 잃었거나 ACK 가 실패했을 수 있다 — 자기 PEL 부터 다시 읽어 순서를 지킨다.
   125	                    // Redis 가 비었으면(재시작 · FLUSHALL) 그룹도 없다 — 다시 만든다(멱등). 회수도 다시 한다.
   126	                    state.backlog = true
   127	                    state.claimDue = true
   128	                    state.groupReady = false
   129	                    state.redisFailures++
   130	                    state.retryAtNanos = System.nanoTime() + backoffNanos(state.redisFailures)
   131	                    log.warn(
   132	                        "persister.redis_failed bucket={} failures={} reason={}",
   133	                        state.bucket, state.redisFailures, e.toString(),
   134	                    )
   135	                }
   136	            }
   137	            if (!progressed) delay(properties.idleDelay.toMillis())
   138	        }
   139	    }
   140	
   141	    /** 버킷 한 걸음 — 메시지를 읽었으면 true. */
   142	    private suspend fun step(state: BucketState): Boolean {
   143	        val b = state.bucket
   144	        if (!state.groupReady) {
   145	            stream.ensureGroup(b)
   146	            state.groupReady = true
   147	        }
   148	        if (state.claimDue) {
   149	            state.claimDue = false
   150	            var cursor = "0-0"
   151	            var claimed = false
   152	            do {
   153	                val page = stream.claimIdle(b, consumer, properties.claimMinIdle, properties.batchSize, cursor)
   154	                if (page.events.isNotEmpty()) claimed = true
   155	                // 멈추면 가져온 것은 이미 내 PEL 이다 — backlog 가 다시 읽는다. 남은 쪽은 다음 회수 때 가져온다
   156	                if (!process(state, page.events)) return true
   157	                cursor = page.cursor
   158	            } while (!page.done)
   159	            if (claimed) return true
   160	        }
   161	        if (state.backlog) {
   162	            val pending = stream.readOwnPending(b, consumer, properties.batchSize)
   163	            if (pending.isNotEmpty()) {
   164	                process(state, pending)
   165	                return true
   166	            }
   167	            state.backlog = false
   168	        }
   169	        val fresh = stream.readNew(b, consumer, properties.batchSize)
   170	        if (fresh.isEmpty()) return false
   171	        process(state, fresh)
   172	        return true
   173	    }
   174	
   175	    /**
   176	     * [events] 를 순서대로 처리하고 처리된 것(dead-letter 로 쌓은 것 포함)을 ACK 한다. 처리기가 실패하고 dead-letter 로도 넘기지 못하면
   177	     * 그 메시지에서 멈추고 false — 버킷은 backoff 뒤 자기 PEL 부터 다시 읽는다.
   178	     */
   179	    private suspend fun process(state: BucketState, events: List<StreamEvent>): Boolean {
   180	        val done = ArrayList<String>(events.size)
   181	        var failed: StreamEvent? = null
   182	        try {
   183	            for (event in events) {
   184	                val outcome = handle(state, event)
   185	                if (outcome is Failed && !(outcome.counted && deadLetter(state, event, outcome))) {
   186	                    failed = event
   187	                    break
   188	                }
   189	                if (state.strikeId == event.id) state.clearStrikes()
   190	                done += event.id
   191	            }
   192	        } finally {
   193	            if (done.isNotEmpty()) {
   194	                if (currentCoroutineContext().isActive) {
   195	                    stream.ack(state.bucket, done) // 실패하면 Redis 오류 — 레인이 backlog 로 다시 읽는다
   196	                } else {
   197	                    // 멈추는 중에도 이미 커밋한 것은 ACK 하되 오래 막히지 않게(못 하면 다시 받아 멱등 SQL 로 같은 결과)
   198	                    withContext(NonCancellable) { withTimeoutOrNull(ACK_ON_STOP_MS) { stream.ack(state.bucket, done) } }
   199	                }
   200	            }
   201	        }
   202	        if (failed == null) {
   203	            state.failures = 0
   204	            return true
   205	        }
   206	        state.failures++
   207	        state.backlog = true
   208	        state.retryAtNanos = System.nanoTime() + backoffNanos(state.failures)
   209	        if (state.failures >= STUCK_AFTER) {
   210	            // 멈춘 버킷은 트림도 멈춰 Stream 이 자란다 — 운영 경보 대상
   211	            log.error("persister.stuck bucket={} id={} type={} failures={}", state.bucket, failed.id, failed.type, state.failures)
   212	        }
   213	        return false
   214	    }
   215	
   216	    /** 메시지 한 건의 결과 — [Done] 이면 ACK 해도 된다(커밋됨 · 건너뜀). */
   217	    private sealed interface Outcome
   218	
   219	    private data object Done : Outcome
   220	
   221	    /** [counted] 면 dead-letter 횟수에 든다. [final] 이면 횟수와 상관없이 바로 dead-letter(다시 해도 같은 형식 오류). */
   222	    private data class Failed(val counted: Boolean, val reason: String, val final: Boolean = false) : Outcome
   223	
   224	    private suspend fun handle(state: BucketState, event: StreamEvent): Outcome {
   225	        val bucket = state.bucket
   226	        if (event.deleted) {
   227	            log.warn("persister.deleted bucket={} id={}", bucket, event.id)
   228	            return Done
   229	        }
   230	        val handler = handlerByType[event.type]
   231	        if (handler == null) {
   232	            // 처리기가 붙기 전 이벤트를 그냥 ACK · 트림으로 잃지 않는다 — 센 실패로 dead-letter 까지 간다(LLD §3.1 처리기 계약)
   233	            log.warn("persister.unhandled bucket={} id={} type={} failures={}", bucket, event.id, event.type, state.failures + 1)
   234	            return Failed(counted = true, reason = "unhandled type: ${event.type}")
   235	        }
   236	        return try {
   237	            handler.handle(event)
   238	            Done
   239	        } catch (e: MalformedEventException) {
   240	            log.error("persister.malformed bucket={} id={} type={} reason={}", bucket, event.id, event.type, e.message)
   241	            Failed(counted = true, reason = e.toString(), final = true)
   242	        } catch (e: Throwable) {
   243	            // 루프가 멈출 때의 취소만 취소다 — 처리기 안의 타임아웃 · Error(TODO() 등)는 처리기 실패로 레인을 살린다
   244	            if (e is CancellationException && !currentCoroutineContext().isActive) throw e
   245	            val counted = PersisterFailures.counts(e)
   246	            log.warn(
   247	                "persister.handler_failed bucket={} id={} type={} failures={} counted={} reason={}",
   248	                bucket, event.id, event.type, state.failures + 1, counted, e.toString(),
   249	            )
   250	            Failed(counted = counted, reason = e.toString())
   251	        }
   252	    }
   253	
   254	    /**
   255	     * 센 실패 하나를 더하고, 문턱에 닿았으면 dead-letter 에 쌓는다(LLD §3.1 dead-letter 2 ~ 4). 쌓았으면 true — 그 메시지를 ACK 한다.
   256	     * 쌓기가 실패하면(DB 장애) false — ACK 하지 않고 다음 시도에서 다시 쌓는다.
   257	     */
   258	    private suspend fun deadLetter(state: BucketState, event: StreamEvent, failure: Failed): Boolean {
   259	        val now = clock.instant()
   260	        if (state.strikeId != event.id) {
   261	            state.strikeId = event.id
   262	            state.strikes = 0
   263	            state.firstStrikeAt = now
   264	        }
   265	        state.strikes++
   266	        if (!failure.final && state.strikes < properties.deadLetterAfter) return false
   267	        val letter = DeadLetter(
   268	            bucket = state.bucket,
   269	            streamId = event.id,
   270	            type = event.type,
   271	            fields = event.fields,
   272	            failureCount = state.strikes,
   273	            lastError = failure.reason,
   274	            firstFailedAt = state.firstStrikeAt ?: now,
   275	            lastFailedAt = now,
   276	        )
   277	        try {
   278	            deadLetters.put(letter)
   279	        } catch (e: Throwable) {
   280	            if (e is CancellationException && !currentCoroutineContext().isActive) throw e
   281	            log.warn(
   282	                "persister.dead_letter_failed bucket={} id={} type={} failures={} reason={}",
   283	                state.bucket, event.id, event.type, state.strikes, e.toString(),
   284	            )
   285	            return false
   286	        }
   287	        log.error(
   288	            "persister.dead_lettered bucket={} id={} type={} failures={} reason={}",
   289	            state.bucket, event.id, event.type, state.strikes, failure.reason,
   290	        )
   291	        state.clearStrikes()
   292	        return true
   293	    }
   294	
   295	    private fun backoffNanos(failures: Int): Long {
   296	        val base = properties.retryBackoff.toNanos()
   297	        val max = properties.retryBackoffMax.toNanos()
   298	        val shift = (failures - 1).coerceIn(0, 30)
   299	        return if (base > max shr shift) max else (base shl shift).coerceAtMost(max)
   300	    }
   301	
   302	    private companion object {
   303	        val log = LoggerFactory.getLogger(StreamPersister::class.java)
   304	
   305	        /** 연속 실패가 이만큼이면 ERROR `persister.stuck`. */
   306	        const val STUCK_AFTER = 10
   307	
   308	        /** 멈추는 중의 ACK 제한. */
   309	        const val ACK_ON_STOP_MS = 5_000L
   310	
   311	        fun handlerMap(handlers: List<PersisterEventHandler>): Map<String, PersisterEventHandler> {
   312	            val byType = HashMap<String, PersisterEventHandler>()
   313	            for (handler in handlers) {
   314	                for (type in handler.types) {
   315	                    val other = byType.putIfAbsent(type, handler)
   316	                    check(other == null) {
   317	                        "persister 처리기가 같은 type 을 둘 맡는다: $type — ${other!!.javaClass.name} · ${handler.javaClass.name}"
   318	                    }
   319	                }
   320	            }
   321	            return byType
   322	        }
   323	    }

     6	/**
     7	 * 처리기 실패가 dead-letter 횟수에 드는지(LLD platform §3.1 dead-letter 1) — DB · Redis · 네트워크 상태로 나는 일시 장애는 세지 않는다.
     8	 *
     9	 * 예외와 원인 사슬의 클래스 계층에 [TRANSIENT] 이름이 하나라도 있으면 일시 장애다. `app:worker` 는 저장소 라이브러리를
    10	 * 컴파일 의존으로 갖지 않아(포트만 쓴다) 클래스 이름으로 본다.
    11	 */
    12	internal object PersisterFailures {
    13	    private val TRANSIENT = setOf(
    14	        // 커넥션 획득 실패 · R2DBC 비일시 자원 오류 · RedisConnectionFailureException
    15	        "org.springframework.dao.DataAccessResourceFailureException",
    16	        // 쿼리 타임아웃 · 일시 자원 오류 · 잠금 · 직렬화 실패
    17	        "org.springframework.dao.TransientDataAccessException",
    18	        "org.springframework.dao.RecoverableDataAccessException",
    19	        "io.r2dbc.spi.R2dbcTransientException",
    20	        "io.r2dbc.spi.R2dbcNonTransientResourceException",
    21	        "java.sql.SQLTransientException",
    22	        "java.sql.SQLRecoverableException",
    23	        "java.io.IOException",
    24	        "java.util.concurrent.TimeoutException",
    25	        "kotlinx.coroutines.TimeoutCancellationException",
    26	    )
    27	
    28	    /** true 면 센다(메시지 탓일 수 있는 실패). */
    29	    fun counts(error: Throwable): Boolean {
    30	        val seen = Collections.newSetFromMap(IdentityHashMap<Throwable, Boolean>())
    31	        var t: Throwable? = error
    32	        while (t != null && seen.add(t)) {
    33	            if (isTransient(t.javaClass)) return false
    34	            t = t.cause
    35	        }
    36	        return true
    37	    }
    38	
    39	    private fun isTransient(type: Class<*>): Boolean {
    40	        var c: Class<*>? = type
    41	        while (c != null) {
    42	            if (c.name in TRANSIENT) return true
    43	            c = c.superclass
    44	        }
    45	        return false
    46	    }
    47	}

     6	/** `worker: persister` 소비 루프 설정(LLD platform §3.1). 인스턴스를 여럿 띄울 때만 [instanceIndex] · [instanceCount] 를 준다. */
     7	@ConfigurationProperties("worker.persister")
     8	data class PersisterProperties(
     9	    /** false 면 루프를 띄우지 않는다(잡만 시험할 때). */
    10	    val enabled: Boolean = true,
    11	    /** 이 인스턴스 번호 — 버킷 `b mod instanceCount == instanceIndex` 를 맡고, 소비자 이름은 `worker-<instanceIndex>`. */
    12	    val instanceIndex: Int = 0,
    13	    val instanceCount: Int = 1,
    14	    /** 레인(코루틴) 수 — 버킷 하나는 레인 하나만 돈다. */
    15	    val parallelism: Int = 16,
    16	    /** 한 번에 읽는 메시지 수(`COUNT` · 1 ~ 1,000). */
    17	    val batchSize: Int = 100,
    18	    /** 한 바퀴 동안 읽은 것이 없을 때 쉬는 시간. */
    19	    val idleDelay: Duration = Duration.ofMillis(200),
    20	    /** `XAUTOCLAIM` 간격 — 기동 직후 한 번, 그 뒤 이 간격마다. */
    21	    val claimInterval: Duration = Duration.ofSeconds(30),
    22	    /** 이보다 오래 ACK 되지 않은 메시지를 회수한다. */
    23	    val claimMinIdle: Duration = Duration.ofSeconds(30),
    24	    /** 처리기 · Redis 실패 뒤 그 버킷을 다시 읽기까지 — 실패가 이어지면 두 배씩 [retryBackoffMax] 까지. */
    25	    val retryBackoff: Duration = Duration.ofSeconds(1),
    26	    val retryBackoffMax: Duration = Duration.ofSeconds(30),
    27	    /** 한 메시지의 센 실패가 이만큼이면 PostgreSQL dead-letter 에 쌓고 ACK 한다(prd D20 · LLD platform §3.1 dead-letter). */
    28	    val deadLetterAfter: Int = 3,
    29	) {

    16	    private val db: DatabaseClient,
    17	) : DeadLetterStore {
    18	    override suspend fun put(letter: DeadLetter) {
    19	        val keys = letter.fields.keys.toList()
    20	        db.sql(
    21	            "INSERT INTO persister_dead_letters " +
    22	                "(bucket, stream_id, event_type, fields, failure_count, last_error, first_failed_at, last_failed_at) " +
    23	                "VALUES (:b, :id, :type, jsonb_object(CAST(:keys AS text[]), CAST(:values AS text[])), :n, :error, :first, :last) " +
    24	                "ON CONFLICT (bucket, stream_id) DO UPDATE SET " +
    25	                "failure_count = GREATEST(persister_dead_letters.failure_count, EXCLUDED.failure_count), " +
    26	                "last_error = EXCLUDED.last_error, " +
    27	                "first_failed_at = LEAST(persister_dead_letters.first_failed_at, EXCLUDED.first_failed_at), " +
    28	                "last_failed_at = GREATEST(persister_dead_letters.last_failed_at, EXCLUDED.last_failed_at)",
    29	        )
    30	            .bind("b", letter.bucket)
    31	            .bind("id", letter.streamId)
    32	            .let { spec -> letter.type?.let { spec.bind("type", it) } ?: spec.bindNull("type", String::class.java) }
    33	            .bind("keys", keys.toTypedArray())
    34	            .bind("values", keys.map { letter.fields.getValue(it) }.toTypedArray())
    35	            .bind("n", letter.failureCount)
    36	            .bind("error", letter.lastError.take(MAX_ERROR))
    37	            .bind("first", odt(letter.firstFailedAt))
    38	            .bind("last", odt(letter.lastFailedAt))
    39	            .fetch().awaitRowsUpdated()
    40	    }
