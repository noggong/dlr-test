+import com.delightroom.mission.core.domain.question.QuestionBank
+import tools.jackson.core.JacksonException
+import tools.jackson.databind.json.JsonMapper

+/**
+ * 은행 키 값 JSON `{"1":[<questions.id>…],"2":[…],"3":[…]}`(LLD game-session §2.3) ↔ [QuestionBank].
+ * 배열 순서 = 응답 순서라 그대로 둔다.
+ */
+class QuestionBankCodec {
+    private val json: JsonMapper = JsonMapper.builder().build()
+
+    fun encode(bank: QuestionBank): String =
+        buildString(bank.idsByDifficulty.values.sumOf { it.size } * 8 + 24) {
+            append('{')
+            Difficulty.entries.forEachIndexed { i, d ->
+                if (i > 0) append(',')
+                append('"').append(d.level).append("\":[")
+                bank.forDifficulty(d).joinTo(this, ",")
+                append(']')
+            }
+            append('}')
+        }
+
+    /** 모양이 틀리면 [IllegalArgumentException]. */
+    fun decode(raw: String): QuestionBank {
+        val root = try {
+            json.readTree(raw)
+        } catch (e: JacksonException) {
+            throw IllegalArgumentException("은행 JSON 을 읽지 못함", e)
+        }
+        require(root.isObject) { "은행 값이 JSON 객체가 아님" }
+        val ids = Difficulty.entries.associateWith { d ->
+            val node = root.get(d.level.toString())
+            require(node != null && node.isArray) { "은행에 난이도 ${d.level} 배열 없음" }
+            List(node.size()) { i ->
+                val id = node.get(i)
+                require(id.isIntegralNumber) { "은행 ID 가 정수가 아님: $id" }
+                id.longValue()
+            }
+        }
+        return QuestionBank(ids)
+    }

+ * 만료는 절대 시각(`PXAT`)이 아니라 게임 시각 [GameClock.now] 부터 `expireAt` 까지 남은 ms 다 — Redis 는 `game.clock-offset` 을 모른다.

+    private val keys: RedisKeys,
+    private val codec: QuestionBankCodec,
+    private val gameClock: GameClock,
+) : QuestionBankCache {
+    override suspend fun get(userId: UserId, date: LocalDate): QuestionBank? =
+        redis.opsForValue().get(keys.bank(userId, date)).awaitSingleOrNull()?.let(codec::decode)
+
+    override suspend fun putIfAbsent(userId: UserId, date: LocalDate, bank: QuestionBank, expireAt: Instant): Boolean =
+        putAllIfAbsent(date, mapOf(userId to bank), expireAt) == 1
+
+    override suspend fun putAllIfAbsent(date: LocalDate, banks: Map<UserId, QuestionBank>, expireAt: Instant): Int {
+        if (banks.isEmpty()) return 0
+        // 08:00 직전 시계 차이로 0 이하가 되면 1 ms — 곧 사라지지만 「없을 때만」은 그대로 지킨다
+        val ttlMs = Duration.between(gameClock.now(), expireAt).toMillis().coerceAtLeast(1L)
+        val expiration = Expiration.from(ttlMs, TimeUnit.MILLISECONDS)
+        return redis.execute { connection ->
+            Flux.fromIterable(banks.entries).flatMapSequential(
+                { (uid, bank) ->
+                    connection.stringCommands().set(
+                        UTF_8.encode(keys.bank(uid, date)),
+                        ByteBuffer.wrap(codec.encode(bank).toByteArray(UTF_8)),
+                        expiration,
+                        SetOption.ifAbsent(),
+                    )
+                },
+                // 묶음 전체를 응답을 기다리지 않고 보낸다
+                banks.size,
+            )
+        }.filter { it }.count().awaitSingle().toInt()
+    }

+-- ARGV[1] uid · ARGV[2] startUs(D 07:00) · ARGV[3] lateUs(D 07:00:03) · ARGV[4] deadlineUs(D 08:00) · ARGV[5] offsetUs
+--      · ARGV[6] 채널 ch:{b}:gid · ARGV[7] D · ARGV[8] gid
+-- 반환(칸은 모두 문자열) — 판정 순서는 SPEC GS14:
+--   { 'ERR', code }                 GAME_NOT_OPEN · SUBMISSION_CLOSED · NOT_PARTICIPANT · SOLO_GROUP
+--   { 'NO_ROSTER' }                 명단이 아직 없다 — 앱이 만들고 한 번 다시 부른다
+--   { 'OK', sa, created, p, st }    created '1' = 이번 호출이 시작시켰다, '0' = 이미 시작함(아무것도 바꾸지 않는다)
+local group, members, ev = KEYS[1], KEYS[2], KEYS[3]
+local uid = ARGV[1]
+local startUs = common.int(ARGV[2], 'startUs')
+local lateUs = common.int(ARGV[3], 'lateUs')
+local deadlineUs = common.int(ARGV[4], 'deadlineUs')
+local t = common.now(ARGV[5])
+local channel, date, gid = ARGV[6], ARGV[7], ARGV[8]
+
+-- 1. 시간 창(GS2) — startUs ≤ t < deadlineUs 만 통과
+if t < startUs then return { 'ERR', 'GAME_NOT_OPEN' } end
+if t >= deadlineUs then return { 'ERR', 'SUBMISSION_CLOSED' } end
+
+-- 2. 명단
+local size = redis.call('HGET', members, '#size')
+if not size then return { 'NO_ROSTER' } end
+
+-- 3. 그날 그룹원(GS3)
+local lid = redis.call('HGET', members, uid)
