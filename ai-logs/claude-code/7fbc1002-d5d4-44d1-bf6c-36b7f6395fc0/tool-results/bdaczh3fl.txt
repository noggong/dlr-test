+            val violations = mutableListOf<String>()
+
+            fun <T> check(parse: () -> T): T? =
+                try {
+                    parse()
+                } catch (e: BusinessException) {

+            }
+            return SignUpForm(id, pw, GroupChoice.of(mode, name))
+        }
