+/** [GroupBoardRepository] 의 Redis 구현 — 판정 · 읽기는 Lua `board_read`(LLD realtime-board §3) 한 번이다. */
+internal class RedisGroupBoardRepository(
+    private val scripts: RedisScripts,
+    private val keys: RedisKeys,
+    private val properties: GameProperties,
+) : GroupBoardRepository {
+    override suspend fun read(groupId: GroupId, date: LocalDate, userId: UserId, mode: BoardReadMode): BoardRead {
+        val (from, to) = window(date, mode)
+        val reply = scripts.eval(
+            "board_read",
+            listOf(keys.members(groupId, date), keys.group(groupId, date)),
+            listOf(
+                userId.value.toString(),
+                from.toString(),
+                to.toString(),
+                properties.clockOffsetUs.toString(),
+                mode.name.lowercase(),
+            ),
+        )
+        return when (reply.firstOrNull()) {
+            "ERR" -> {
+                check(reply.size == 2) { "board_read 거부 반환은 {ERR, code} 다: $reply" }
+                val code = ErrorCode.valueOf(reply[1] as String)
+                check(code in REJECTIONS) { "board_read 가 모르는 거부 코드를 돌려줬다: $code" }
+                BoardRead.Rejected(code)
+            }
+            "OK" -> {
+                check(reply.size >= 2 && reply.size % 2 == 0) { "board_read 통과 반환은 {OK, now, 필드, 값…} 이다: $reply" }
+                BoardRead.Opened(microsToInstant((reply[1] as String).toLong()), reply.drop(2).map { it as String })
+            }
+            else -> error("board_read 반환의 첫 칸은 OK · ERR 이다: $reply")
+        }
+    }
+
+    /** 모드별 창 [시작, 끝) µs — LLD §3 표. */
+    private fun window(date: LocalDate, mode: BoardReadMode): Pair<Long, Long> {
+        val start = date.atTime(properties.startTime).atZone(properties.zone).toInstant()
+        return when (mode) {
+            BoardReadMode.BOARD -> micros(start) to micros(properties.deadlineAt(date))
+            BoardReadMode.TICKET -> micros(start) to micros(properties.graceEndAt(date))
+            BoardReadMode.STREAM -> 0L to Long.MAX_VALUE
+        }
+    }
+
+    private fun micros(at: Instant): Long = Math.addExact(Math.multiplyExact(at.epochSecond, 1_000_000L), at.nano / 1_000L)
+
+    private companion object {
+        val REJECTIONS = setOf(ErrorCode.GAME_NOT_OPEN, ErrorCode.NOT_PARTICIPANT, ErrorCode.SOLO_GROUP)
+    }
+}

+-- 현황판 읽기(LLD realtime-board §3 · SPEC RB1 · RB2 · RB3) — 조회 · 티켓 · 스트림이 같은 판정 순서로 쓴다.
+-- KEYS[1] 명단 Hash g:{b}:gid:date:members · KEYS[2] 그룹 Hash g:{b}:gid:date
+-- ARGV[1] uid · ARGV[2] 창 시작 µs · ARGV[3] 창 끝 µs(미만까지) · ARGV[4] offsetUs · ARGV[5] 모드 board · ticket · stream
+-- 반환 { 'ERR', code } 또는 { 'OK', now } (ticket) · { 'OK', now, HGETALL… } — now 는 µs 십진 문자열.
+-- 처음 걸린 판정으로 끝낸다. 아무 키도 쓰지 않는다.
+local roster, group = KEYS[1], KEYS[2]
+local uid, mode = ARGV[1], ARGV[5]
+if mode ~= 'board' and mode ~= 'ticket' and mode ~= 'stream' then
+  error('board_read: mode must be board, ticket or stream: ' .. tostring(mode))
+end
+local from = common.int(ARGV[2], 'board_read: window start')
+local to = common.int(ARGV[3], 'board_read: window end')
+local t = common.now(ARGV[4])
+
+-- 1. 시간 창
+if t < from or t >= to then return { 'ERR', 'GAME_NOT_OPEN' } end
+-- 2. 명단 확정 전
+local size = redis.call('HGET', roster, '#size')
+if not size then return { 'ERR', 'GAME_NOT_OPEN' } end
+-- 3. 그날 그룹원 아님(07:00 이후 가입)
+if redis.call('HEXISTS', roster, uid) == 0 then return { 'ERR', 'NOT_PARTICIPANT' } end
+-- 4. 1명 그룹
+if common.int(size, 'board_read: #size') < 2 then return { 'ERR', 'SOLO_GROUP' } end
+-- 5. 티켓은 그룹 Hash 를 읽지 않는다
+local now = common.fmt(t)
+if mode == 'ticket' then return { 'OK', now } end
+-- 6. 그룹 Hash 없음(재적재 전)
+if redis.call('EXISTS', group) == 0 then return { 'ERR', 'GAME_NOT_OPEN' } end
+-- 7.
+local reply = { 'OK', now }
+for _, v in ipairs(redis.call('HGETALL', group)) do reply[#reply + 1] = v end
+return reply
