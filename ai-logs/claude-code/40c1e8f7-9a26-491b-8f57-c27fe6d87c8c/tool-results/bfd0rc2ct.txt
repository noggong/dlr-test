    42	        val kinds = rows(
    43	            "SELECT c.relname, c.relkind::text FROM pg_class c JOIN pg_namespace n ON n.oid = c.relnamespace " +
    44	                "WHERE n.nspname = 'public' AND c.relname::text = ANY(:names)",

   123	    fun `O14 난이도마다 120개이고 분야 4종이 다 있다`() {
   124	        val perDifficulty = rows("SELECT difficulty, count(*) FROM questions GROUP BY difficulty") {

   131	        }
   132	        assertThat(categories.map { it.second }.toSet()).isEqualTo(CATEGORIES)
   133	        (1..3).forEach { d -> assertThat(categories.filter { it.first == d }.map { it.second }.toSet()).isEqualTo(CATEGORIES) }

   136	    @Test
   137	    fun `O14 pk_hash 는 겹치지 않고 answer_hash 는 prd D7 식과 같다`() {
   138	        val questions = rows("SELECT pk_hash, answer, answer_hash FROM questions") {
   139	            Triple(it.get(0, String::class.java)!!, it.get(1, String::class.java)!!, it.get(2, String::class.java)!!)
   140	        }
   141	        assertThat(questions).hasSize((SEED_PER_DIFFICULTY * 3).toInt())
   142	        assertThat(questions.map { it.first }).doesNotHaveDuplicates().allMatch { HEX64.matches(it) }
   143	        questions.forEach { (pk, answer, hash) ->
   144	            assertThat(hash).describedAs(pk).matches(HEX64.pattern).isEqualTo(sha256Hex("$pk:${normalize(answer)}"))
   145	        }
   146	        // 정규화가 실제로 일을 한다(공백 · 대문자가 있는 답) — 원문 그대로 해시하면 달라야 한다
   147	        assertThat(questions.first().let { (pk, answer, hash) -> sha256Hex("$pk:$answer") != hash }).isTrue()
   148	    }

   152	    /** prd R-A2 — 띄어쓰기 전부 제거 → 소문자 → 앞뒤 공백 제거. */
   153	    private fun normalize(answer: String): String = answer.filterNot { it.isWhitespace() }.lowercase().trim()
   154	
   155	    private fun sha256Hex(s: String): String =
   156	        MessageDigest.getInstance("SHA-256").digest(s.toByteArray(Charsets.UTF_8)).joinToString("") { "%02x".format(it) }
   157	
   158	    private fun snapshot(): Map<String, Long> = mapOf(
   159	        "tables" to count("SELECT count(*) FROM pg_class WHERE relnamespace = 'public'::regnamespace AND relkind IN ('r', 'p')"),
   160	        "indexes" to count("SELECT count(*) FROM pg_indexes WHERE schemaname = 'public'"),
   161	        "partitions" to count("SELECT count(*) FROM pg_inherits"),
   162	        "questions" to count("SELECT count(*) FROM questions"),
   163	    )
   164	
   165	    private fun count(sql: String): Long = rows(sql) { (it.get(0) as Number).toLong() }.single()
   166	
   167	    private fun single(sql: String): String = strings(sql).single()
   168	
   169	    /** 첫 칸을 문자열로 — SQL NULL 은 [NULL] 로 바꾼다(R2DBC 는 null 행을 못 낸다). */
   170	    private fun strings(sql: String, vararg binds: Pair<String, Any>): List<String> =
   171	        rows(sql, *binds) { it.get(0, String::class.java) ?: NULL }
   172	
   173	    private fun <T : Any> rows(sql: String, vararg binds: Pair<String, Any>, map: (Readable) -> T): List<T> =
   174	        runBlocking {
   175	            binds.fold(db.sql(sql)) { spec, (k, v) -> spec.bind(k, v) }
   176	                .map { row -> map(row) }
   177	                .all().collectList().awaitSingle()
   178	        }

   182	        val TABLES = listOf("groups", "users", "questions", "served_questions") + DAILY_TABLES
   183	        val INDEXES = mapOf(
   184	            "users_group_joined_idx" to "(group_id, joined_at)",
   185	            "questions_difficulty_idx" to "(difficulty)",
   186	            "daily_progress_group_idx" to "(game_date, group_id)",
   187	            "group_results_group_idx" to "(group_id, game_date DESC)",
   188	            "group_result_members_group_idx" to "(game_date, group_id)",
   189	        )
   190	        val CATEGORIES = setOf("POLITICS", "CURRENT", "SOCIETY", "GENERAL")
   191	        const val SEED_PER_DIFFICULTY = 120L
   192	        const val NULL = "<null>"
   193	        val HEX64 = Regex("^[0-9a-f]{64}$")
   194	    }
