+package com.delightroom.mission.app.api.auth

+import com.delightroom.mission.core.config.AuthProperties
+import com.delightroom.mission.core.domain.auth.IssuedTokens
+import java.time.Duration
+import java.time.Instant
+import java.time.ZoneOffset
+import java.time.format.DateTimeFormatter
+import java.util.Locale
+
+/**
+ * `access` · `refresh` 쿠키의 `Set-Cookie` 값(auth-signup SPEC §2 쿠키 표).
+ *
+ * Spring `ResponseCookie` 는 `Expires` 를 시스템 시계 + `Max-Age` 로 쓴다 — 앱의 `Clock` · `game.clock-offset` 과
+ * 어긋나므로 `Expires` 를 토큰 `exp`(= 본문 `accessExpiresAt` · `refreshExpiresAt`)로 직접 적는다.
+ * 속성 순서는 SPEC 예시와 같다: `Path` · `Max-Age` · `Expires` · `Secure` · `HttpOnly` · `SameSite`.
+ */
+class AuthCookies(private val properties: AuthProperties) {
+    /** 로그인 · refresh `200` 의 두 쿠키. */
+    fun issue(tokens: IssuedTokens): List<String> =
+        listOf(
+            cookie(ACCESS, tokens.access, ACCESS_PATH, properties.accessTtl, tokens.accessExpiresAt),
+            cookie(REFRESH, tokens.refresh, REFRESH_PATH, properties.refreshTtl, tokens.refreshExpiresAt),
+        )
+
+    /** 로그아웃 `204` 의 지우기 쿠키 두 개(값 빈 문자열 · `Max-Age=0` · 1970 `Expires`). */
+    fun clear(): List<String> =
+        listOf(
+            cookie(ACCESS, "", ACCESS_PATH, Duration.ZERO, Instant.EPOCH),
+            cookie(REFRESH, "", REFRESH_PATH, Duration.ZERO, Instant.EPOCH),
+        )
+
+    private fun cookie(name: String, value: String, path: String, maxAge: Duration, expires: Instant): String =
+        buildString {
+            append(name).append('=').append(value)
+            append("; Path=").append(path)
+            append("; Max-Age=").append(maxAge.seconds)
+            append("; Expires=").append(HTTP_DATE.format(expires))
+            if (properties.cookieSecure) append("; Secure")
+            append("; HttpOnly; SameSite=Lax")
+        }
+
+    companion object {
+        const val ACCESS = "access"
+        const val REFRESH = "refresh"
+        const val ACCESS_PATH = "/api"
+        const val REFRESH_PATH = "/api/v1/auth"
+
+        /** RFC 7231 IMF-fixdate — 일은 두 자리(`01 Jan`). `RFC_1123_DATE_TIME` 은 한 자리라 쓰지 않는다. */
+        private val HTTP_DATE: DateTimeFormatter =
+            DateTimeFormatter.ofPattern("EEE, dd MMM yyyy HH:mm:ss 'GMT'", Locale.US).withZone(ZoneOffset.UTC)
+    }

+package com.delightroom.mission.app.api.auth
+
+import com.delightroom.mission.core.domain.auth.RawPassword
+import com.delightroom.mission.core.port.PasswordHasher
+import kotlinx.coroutines.Dispatchers
+import kotlinx.coroutines.withContext
+import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder
+
+/**
+ * BCrypt(`$2a$<cost>$` — auth-signup LLD §2.1). cost 10 은 한 번에 약 50~100ms 라 이벤트 루프가 아니라
+ * `Dispatchers.Default` 에서 돌린다.
+ */
+class BcryptPasswordHasher(cost: Int) : PasswordHasher {
+    private val encoder = BCryptPasswordEncoder(cost)
+
+    /** [dummyMatch] 가 비교할 해시 — 같은 cost 로 만들어 있는 아이디의 [matches] 와 시간이 같다. */
+    private val dummyHash: String = encoder.encode(DUMMY_PASSWORD)!!
+
+    override suspend fun hash(raw: RawPassword): String =
+        withContext(Dispatchers.Default) { encoder.encode(raw.value)!! }
+
+    override suspend fun matches(raw: RawPassword, hash: String): Boolean =
+        withContext(Dispatchers.Default) { encoder.matches(raw.value, hash) }
+
+    override suspend fun dummyMatch() {
+        withContext(Dispatchers.Default) { encoder.matches(DUMMY_COMPARE, dummyHash) }
+    }
+
+    private companion object {
+        const val DUMMY_PASSWORD = "timing-dummy-password"
+        const val DUMMY_COMPARE = "timing-dummy-compare!"
+    }

+/**
+ * JWS HS256 토큰(auth-signup LLD §2.3). 헤더는 `{"alg":"HS256"}` 하나.
+ *
+ * - access = `sub`(users.id) · `lid` · `gid` · `iat` · `exp`(+[AuthProperties.accessTtl])
+ * - refresh = `sub` · `jti` · `iat` · `exp`(+[AuthProperties.refreshTtl])
+ *
+ * 두 토큰은 클레임 모양으로 갈린다 — refresh 를 access 로(또는 반대로) 넣으면 검증이 `null` 이다.
+ */
+class JwtTokenIssuer(private val properties: AuthProperties) : TokenIssuer {
+    private val secret = properties.jwtSecret.toByteArray(Charsets.UTF_8)
+    private val signer = MACSigner(secret)
+    private val verifier = MACVerifier(secret)
+
+    override fun issue(account: Account, jti: String, now: Instant): IssuedTokens {
+        val iat = now.truncatedTo(ChronoUnit.SECONDS)
+        val accessExp = iat.plus(properties.accessTtl)
+        val refreshExp = iat.plus(properties.refreshTtl)
+        val access = JWTClaimsSet.Builder()
+            .subject(account.userId.value.toString())
+            .claim(LID, account.loginId.value)
+            .claim(GID, account.groupId.value.toString())
+            .issueTime(Date.from(iat))
+            .expirationTime(Date.from(accessExp))
+            .build()
+        val refresh = JWTClaimsSet.Builder()
+            .subject(account.userId.value.toString())
+            .jwtID(jti)
+            .issueTime(Date.from(iat))
+            .expirationTime(Date.from(refreshExp))
+            .build()
+        return IssuedTokens(sign(access), accessExp, sign(refresh), refreshExp)
+    }
+
+    override fun verifyAccess(token: String, now: Instant): AccessClaims? {
+        val claims = verified(token, now) ?: return null
+        if (claims.jwtid != null) return null
+        return parseOrNull {
+            AccessClaims(
+                userId = UserId(claims.subject?.toLongOrNull() ?: return null),
+                loginId = LoginId(claims.getStringClaim(LID) ?: return null),
+                groupId = GroupId(claims.getStringClaim(GID)?.toLongOrNull() ?: return null),
+                expiresAt = claims.expirationTime.toInstant(),
+            )
+        }
+    }
+
+    override fun verifyRefresh(token: String, now: Instant): RefreshClaims? {
+        val claims = verified(token, now) ?: return null
+        if (claims.claims.containsKey(LID)) return null
+        return RefreshClaims(
+            userId = UserId(claims.subject?.toLongOrNull() ?: return null),
+            jti = claims.jwtid ?: return null,
+            expiresAt = claims.expirationTime.toInstant(),
+        )
+    }
+
+    private fun sign(claims: JWTClaimsSet): String =
+        SignedJWT(HEADER, claims).apply { sign(signer) }.serialize()
+
+    /** 서명 · 알고리즘 · 만료(`now < exp`)를 통과한 클레임. 아니면 `null`. */
+    private fun verified(token: String, now: Instant): JWTClaimsSet? =
+        parseOrNull {
+            val jwt = SignedJWT.parse(token)
+            if (jwt.header.algorithm != JWSAlgorithm.HS256 || !jwt.verify(verifier)) return null
+            val claims = jwt.jwtClaimsSet
+            val exp = claims.expirationTime ?: return null
+            if (!now.isBefore(exp.toInstant())) return null
+            claims
+        }
+
+    /** 모양이 틀린 토큰(JWS 가 아님 · 클레임 타입이 다름)은 `null` 로 — 401 한 갈래로 간다. */
+    private inline fun <T> parseOrNull(block: () -> T): T? =
+        try {
+            block()
+        } catch (_: ParseException) {
+            null
+        } catch (_: JOSEException) {
+            null
+        }
+
+    private companion object {
+        const val LID = "lid"
+        const val GID = "gid"
+        val HEADER: JWSHeader = JWSHeader(JWSAlgorithm.HS256)
+    }
+}

+    override suspend fun start(userId: UserId, jti: String, ttl: Duration): Boolean =
+        redis.opsForValue().setIfAbsent(keys.session(userId), jti, ttl.checkSeconds()).awaitSingle()
+
+    override suspend fun rotate(userId: UserId, oldJti: String, newJti: String, ttl: Duration): Boolean =
+        scripts.eval(
+            "sess_rotate",
+            listOf(keys.session(userId)),
+            listOf(oldJti, newJti, ttl.checkSeconds().seconds.toString()),
+        ).single() == 1L
+
+    override suspend fun end(userId: UserId, jti: String): Boolean =
+        scripts.eval("sess_end", listOf(keys.session(userId)), listOf(jti)).single() == 1L

+    private fun Duration.checkSeconds(): Duration {
+        require(seconds > 0 && nano == 0) { "세션 TTL 은 양의 초 단위다: $this" }
+        return this
+    }
