    13	 *
    14	 * Spring `ResponseCookie` 는 `Expires` 를 시스템 시계 + `Max-Age` 로 쓴다 — 앱의 `Clock` · `game.clock-offset` 과
    15	 * 어긋나므로 `Expires` 를 토큰 `exp`(= 본문 `accessExpiresAt` · `refreshExpiresAt`)로 직접 적는다.
    16	 * 속성 순서는 SPEC 예시와 같다: `Path` · `Max-Age` · `Expires` · `Secure` · `HttpOnly` · `SameSite`.
    17	 */
    18	class AuthCookies(private val properties: AuthProperties) {
    19	    /** 로그인 · refresh `200` 의 두 쿠키. */
    20	    fun issue(tokens: IssuedTokens): List<String> =
    21	        listOf(
    22	            cookie(ACCESS, tokens.access, ACCESS_PATH, properties.accessTtl, tokens.accessExpiresAt),
    23	            cookie(REFRESH, tokens.refresh, REFRESH_PATH, properties.refreshTtl, tokens.refreshExpiresAt),
    24	        )
    25	
    26	    /** 로그아웃 `204` 의 지우기 쿠키 두 개(값 빈 문자열 · `Max-Age=0` · 1970 `Expires`). */
    27	    fun clear(): List<String> =
    28	        listOf(
    29	            cookie(ACCESS, "", ACCESS_PATH, Duration.ZERO, Instant.EPOCH),
    30	            cookie(REFRESH, "", REFRESH_PATH, Duration.ZERO, Instant.EPOCH),
    31	        )
    32	
    33	    private fun cookie(name: String, value: String, path: String, maxAge: Duration, expires: Instant): String =
    34	        buildString {
    35	            append(name).append('=').append(value)
    36	            append("; Path=").append(path)
    37	            append("; Max-Age=").append(maxAge.seconds)
    38	            append("; Expires=").append(HTTP_DATE.format(expires))
    39	            if (properties.cookieSecure) append("; Secure")
    40	            append("; HttpOnly; SameSite=Lax")
    41	        }
    42	
    43	    companion object {
    44	        const val ACCESS = "access"
    45	        const val REFRESH = "refresh"
    46	        const val ACCESS_PATH = "/api"
    47	        const val REFRESH_PATH = "/api/v1/auth"
    48	
    49	        /** RFC 7231 IMF-fixdate — 일은 두 자리(`01 Jan`). `RFC_1123_DATE_TIME` 은 한 자리라 쓰지 않는다. */
    50	        private val HTTP_DATE: DateTimeFormatter =
    51	            DateTimeFormatter.ofPattern("EEE, dd MMM yyyy HH:mm:ss 'GMT'", Locale.US).withZone(ZoneOffset.UTC)
    52	    }

    20	        withContext(Dispatchers.Default) { encoder.encode(raw.value)!! }
    21

    24	
    25	    override suspend fun dummyMatch() {
    26	        withContext(Dispatchers.Default) { encoder.matches(DUMMY_COMPARE, dummyHash) }
    27	    }

    24	/**
    25	 * JWS HS256 토큰(auth-signup LLD §2.3). 헤더는 `{"alg":"HS256"}` 하나.
    26	 *
    27	 * - access = `sub`(users.id) · `lid` · `gid` · `iat` · `exp`(+[AuthProperties.accessTtl])
    28	 * - refresh = `sub` · `jti` · `iat` · `exp`(+[AuthProperties.refreshTtl])
    29	 *
    30	 * 두 토큰은 클레임 모양으로 갈린다 — refresh 를 access 로(또는 반대로) 넣으면 검증이 `null` 이다.
    31	 */
    32	class JwtTokenIssuer(private val properties: AuthProperties) : TokenIssuer {
    33	    private val secret = properties.jwtSecret.toByteArray(Charsets.UTF_8)
    34	    private val signer = MACSigner(secret)
    35	    private val verifier = MACVerifier(secret)
    36	
    37	    override fun issue(account: Account, jti: String, now: Instant): IssuedTokens {
    38	        val iat = now.truncatedTo(ChronoUnit.SECONDS)
    39	        val accessExp = iat.plus(properties.accessTtl)
    40	        val refreshExp = iat.plus(properties.refreshTtl)
    41	        val access = JWTClaimsSet.Builder()
    42	            .subject(account.userId.value.toString())
    43	            .claim(LID, account.loginId.value)
    44	            .claim(GID, account.groupId.value.toString())
    45	            .issueTime(Date.from(iat))
    46	            .expirationTime(Date.from(accessExp))
    47	            .build()
    48	        val refresh = JWTClaimsSet.Builder()
    49	            .subject(account.userId.value.toString())
    50	            .jwtID(jti)
    51	            .issueTime(Date.from(iat))
    52	            .expirationTime(Date.from(refreshExp))
    53	            .build()
    54	        return IssuedTokens(sign(access), accessExp, sign(refresh), refreshExp)
    55	    }
    56	
    57	    override fun verifyAccess(token: String, now: Instant): AccessClaims? {
    58	        val claims = verified(token, now) ?: return null
    59	        if (claims.jwtid != null) return null
    60	        return parseOrNull {
    61	            AccessClaims(
    62	                userId = UserId(claims.subject?.toLongOrNull() ?: return null),
    63	                loginId = LoginId(claims.getStringClaim(LID) ?: return null),
    64	                groupId = GroupId(claims.getStringClaim(GID)?.toLongOrNull() ?: return null),
    65	                expiresAt = claims.expirationTime.toInstant(),
    66	            )
    67	        }
    68	    }
    69	
    70	    override fun verifyRefresh(token: String, now: Instant): RefreshClaims? {
    71	        val claims = verified(token, now) ?: return null
    72	        if (claims.claims.containsKey(LID)) return null
    73	        return RefreshClaims(
    74	            userId = UserId(claims.subject?.toLongOrNull() ?: return null),
    75	            jti = claims.jwtid ?: return null,
    76	            expiresAt = claims.expirationTime.toInstant(),
    77	        )
    78	    }
    79	
    80	    private fun sign(claims: JWTClaimsSet): String =
    81	        SignedJWT(HEADER, claims).apply { sign(signer) }.serialize()
    82	
    83	    /** 서명 · 알고리즘 · 만료(`now < exp`)를 통과한 클레임. 아니면 `null`. */
    84	    private fun verified(token: String, now: Instant): JWTClaimsSet? =
    85	        parseOrNull {
    86	            val jwt = SignedJWT.parse(token)
    87	            if (jwt.header.algorithm != JWSAlgorithm.HS256 || !jwt.verify(verifier)) return null
    88	            val claims = jwt.jwtClaimsSet
    89	            val exp = claims.expirationTime ?: return null
    90	            if (!now.isBefore(exp.toInstant())) return null
    91	            claims
    92	        }
    93	
    94	    /** 모양이 틀린 토큰(JWS 가 아님 · 클레임 타입이 다름)은 `null` 로 — 401 한 갈래로 간다. */
    95	    private inline fun <T> parseOrNull(block: () -> T): T? =
    96	        try {
    97	            block()
    98	        } catch (_: ParseException) {
    99	            null
   100	        } catch (_: JOSEException) {
   101	            null
   102	        }
   103	
   104	    private companion object {
   105	        const val LID = "lid"
   106	        const val GID = "gid"
   107	        val HEADER: JWSHeader = JWSHeader(JWSAlgorithm.HS256)
   108	    }
   109	}
