   215	        /** 게임 날짜 KST 시각의 채널 메시지 하나를 `SPUBLISH` 한다 — 그룹 Hash 는 [hash] 에 [version] 을 덮은 것(LLD §2.3). */
   216	        private fun publish(
   217	            type: GroupChannelType,
   218	            version: Long,
   219	            hash: Map<String, String> = BoardFixtures.inProgressHash(),
   220	            uid: Long? = BoardFixtures.JUNHO,
   221	            at: String = "07:31:02.004118",
   222	        ) {
   223	            val raw = codec.encode(
   224	                type, gid, DATE, "$version-0", uid?.let(::UserId), version, BoardFixtures.micros(LocalTime.parse(at)),
   225	                hash + ("version" to "$version"),
   226	            )
   227	            redis.spublish(keys.channel(gid), raw)
   228	        }

   230	        /** 스트림을 열어 둔 채 프레임을 하나씩 받는다. 끝나면(정상 · 오류) [ended] 가 풀린다. */
   231	        private class Frames(flux: Flux<ServerSentEvent<String>>) {
   232	            private val queue = LinkedBlockingQueue<ServerSentEvent<String>>()
   233	            private val ended = CountDownLatch(1)
   234	
   235	            @Volatile var error: Throwable? = null
   236	            private val subscription = flux.subscribe({ queue.add(it) }, { error = it; ended.countDown() }, { ended.countDown() })
   237	
   238	            fun next(): ServerSentEvent<String> =
   239	                checkNotNull(queue.poll(TIMEOUT.toMillis(), TimeUnit.MILLISECONDS)) { "5초 안에 프레임이 오지 않음" }
   240	
   241	            /** [wait] 동안 더 온 프레임이 없다. */
   242	            fun noMore(wait: Duration = Duration.ofMillis(500)): Boolean = queue.poll(wait.toMillis(), TimeUnit.MILLISECONDS) == null
   243	
   244	            fun endedWithin(wait: Duration): Boolean = ended.await(wait.toMillis(), TimeUnit.MILLISECONDS)
   245	
   246	            fun close() = subscription.dispose()
   247	        }
   248	
   249	        /** 연결해 첫 프레임(snapshot)까지 받은 스트림 — 이 뒤 발행한 메시지는 반드시 이 연결에 온다(구독이 읽기보다 먼저 — RB5). */
   250	        private fun open(uid: Long = MINJI): Pair<Frames, ServerSentEvent<String>> {
   251	            val frames = Frames(stream(connect(ticket(uid))).responseBody)
   252	            val first = frames.next()
   253	            assertThat(first.event()).isEqualTo("board.snapshot")
   254	            return frames to first
   255	        }

   257	        @Suppress("UNCHECKED_CAST")
   258	        private fun entries(frame: ServerSentEvent<String>): List<Map<String, Any?>> = data(frame)["entries"] as List<Map<String, Any?>>
   259	
   260	        /**
   261	         * 현황판을 실은 모든 프레임의 불변식 — I5 결과 없는 그룹에서 `CHEAT_FAILED` 줄은 보는 사람 본인뿐 · I6 명단 전원 한 번씩 ·
   262	         * I7 경쟁 순위 · `id` = `data.version`(RB15).
   263	         */
   264	        private fun assertInvariants(frame: ServerSentEvent<String>, viewer: String) {
   265	            val data = data(frame)
   266	            assertThat(frame.id()).isEqualTo("${data["version"]}")
   267	            if ("entries" !in data) return
   268	            val rows = entries(frame)
   269	            assertThat(rows.map { it["userId"] }).hasSize(MEMBERS.size).doesNotHaveDuplicates()
   270	                .containsExactlyInAnyOrderElementsOf(MEMBERS.values)
   271	            if (data["groupStatus"] in setOf("IN_PROGRESS", "PENDING")) {
   272	                assertThat(rows.filter { it["status"] == "CHEAT_FAILED" }.map { it["userId"] }).isSubsetOf(viewer)
   273	            }
   274	            val ranks = rows.map { it["rank"] as Int }
   275	            assertThat(ranks.first()).isEqualTo(1)
   276	            ranks.forEach { r -> assertThat(r).isEqualTo(ranks.count { it < r } + 1) }
   277	        }
   278	
   279	        private fun row(frame: ServerSentEvent<String>, userId: String): Map<String, Any?> = entries(frame).single { it["userId"] == userId }
