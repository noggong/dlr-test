+/**
+ * 게임 날짜 고르기(SPEC GS1)와 그 날짜의 경계 시각(LLD game-session §1).
+ * 경계 시각은 [Instant] 와 Lua ARGV 용 epoch µs 로 둘 다 준다 — Lua 의 `t` 는 Redis `TIME`(µs)이다.

+class GameClock(
+    private val properties: GameProperties,
+    private val clock: Clock,
+) {
+    /** 지금(시각 주입 `game.clock-offset` 포함 — [GameProperties.now]). */
+    fun now(): Instant = properties.now(clock)
+
+    /** 게임 정보의 날짜(GS1): KST 로 그 날짜 08:00:00 전이면 그 날짜, 08:00:00 부터 다음 날. */
+    fun infoDate(now: Instant): LocalDate {
+        val date = playDate(now)
+        return if (now < deadlineAt(date)) date else date.plusDays(1)
+    }
+
+    /** 문제은행 · 진입 신호의 날짜(GS1): 서버 시각의 KST 날짜. */
+    fun playDate(now: Instant): LocalDate = now.atZone(properties.zone).toLocalDate()

+
+    /** 07:00:03 — 이 시각보다 늦게(`>`) 시작하면 지각(D8). */

+
+    /** 08:00:00 — 마감. 은행 키 `PXAT` 도 이 시각이다(LLD §2.1). */
+    fun deadlineAt(date: LocalDate): Instant = properties.deadlineAt(date)

+sealed interface BankDraw {
+    data class Built(val bank: QuestionBank) : BankDraw

+    /** [difficulty] 에 이 유저의 미출제 문제가 0개 — 은행을 만들지 않는다(SPEC GS6 · `QUESTIONS_EXHAUSTED`). */

+/**
+ * 은행 뽑기(SPEC GS6 · prd D10 · LLD §4.4) — 순수 함수. `api` 대체 경로와 `worker: prepare-questions` 가 같이 쓴다.
+ * 난이도마다 후보 = 문제 캐시 − 출제 이력, 후보가 [size] 개 이하면 전부(섞어서), 넘으면 부분 Fisher–Yates 로 [size] 개.
+ */
+object QuestionBankBuilder {
+    fun build(
+        idsByDifficulty: Map<Difficulty, LongArray>,
+        served: Set<Long>,
+        size: Int,
+        random: Random = ThreadLocalRandom.current().asKotlinRandom(),
+    ): BankDraw {
+        require(size > 0) { "size 는 1 이상이다: $size" }
+        val bank = LinkedHashMap<Difficulty, List<Long>>()
+        for (d in Difficulty.entries) {
+            val candidates = idsByDifficulty[d]?.filterNotTo(ArrayList()) { it in served } ?: arrayListOf()
+            if (candidates.isEmpty()) return BankDraw.Exhausted(d)
+            bank[d] = pick(candidates, size, random)
+        }
+        return BankDraw.Built(QuestionBank(bank))
+    }
+
+    /** 부분 Fisher–Yates — 앞 `min(size, n)` 칸만 섞어 그만큼 자른다. [candidates] 를 바꾼다. */
+    private fun pick(candidates: ArrayList<Long>, size: Int, random: Random): List<Long> {
+        val n = minOf(size, candidates.size)
+        for (i in 0 until n) {
+            val j = random.nextInt(i, candidates.size)
+            candidates[i] = candidates[j].also { candidates[j] = candidates[i] }
+        }
+        return candidates.subList(0, n).toList()
+    }

@Import(MutableClockConfiguration::class)
@ContextConfiguration(initializers = [PostgresTestContainer.Initializer::class])

    @Autowired
    private lateinit var catalog: QuestionCatalog

    @Autowired
    private lateinit var db: DatabaseClient

    private data class Row(val id: Long, val pk: String, val difficulty: Int, val category: String, val body: String, val answerHash: String)

    private fun rows(): List<Row> = runBlocking {
        db.sql("SELECT id, pk_hash, difficulty, category, body, answer_hash FROM questions ORDER BY id")
            .map { r ->
                Row(
                    r.get("id", Number::class.java)!!.toLong(),
                    r.get("pk_hash", String::class.java)!!,
                    r.get("difficulty", Number::class.java)!!.toInt(),
                    r.get("category", String::class.java)!!,
                    r.get("body", String::class.java)!!,
                    r.get("answer_hash", String::class.java)!!,
                )
            }
            .flow()
            .toList()
    }

    @Test
    fun `O14 기동 때 questions 전체가 문제 캐시에 오른다 — 시드 난이도마다 120개 이상`() {
        val rows = rows()

        for (d in Difficulty.entries) {

        }
    }

    @Test
    fun `O13 캐시의 PK · 난이도 · 분야 · 본문 · answerHash 가 questions 행과 같다`() {
        for (row in rows()) {

            assertThat(q.id.value).isEqualTo(row.pk).hasSize(64)
            assertThat(catalog.byPk.getValue(q.id)).isSameAs(q)
            assertThat(q.difficulty.level).isEqualTo(row.difficulty)
            assertThat(q.category.name).isEqualTo(row.category)
            assertThat(q.body).isEqualTo(row.body)
            assertThat(q.answerHash).isEqualTo(row.answerHash).hasSize(64)
        }
