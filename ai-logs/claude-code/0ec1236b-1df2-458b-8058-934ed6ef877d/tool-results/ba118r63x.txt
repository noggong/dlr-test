    29	    private val tokenIssuer: TokenIssuer,
    30	    private val gameProperties: GameProperties,
    31	    private val clock: Clock,
    32	) : WebFilter, Ordered {
    33	    override fun getOrder(): Int = ORDER
    34	
    35	    override fun filter(exchange: ServerWebExchange, chain: WebFilterChain): Mono<Void> {
    36	        val request = exchange.request
    37	        if (!requiresAuth(request.method, request.path.pathWithinApplication())) return chain.filter(exchange)
    38	
    39	        val token = request.cookies.getFirst(AuthCookies.ACCESS)?.value
    40	        val claims = token?.takeIf { it.isNotEmpty() }?.let { tokenIssuer.verifyAccess(it, gameProperties.now(clock)) }
    41	            ?: return unauthorized(exchange)
    42	
    43	        exchange.attributes[AuthPrincipal.ATTRIBUTE] = AuthPrincipal(claims.userId, claims.loginId, claims.groupId)
    44	        return chain.filter(exchange)
    45	    }

    48	        val response = exchange.response
    49	        response.statusCode = HttpStatus.valueOf(ErrorCode.UNAUTHORIZED.httpStatus)

    51	        return response.writeWith(Mono.fromSupplier { response.bufferFactory().wrap(UNAUTHORIZED_BODY) })
    52	    }

    28	 *
    29	 * - 「지금」은 [GameClock.now](`Clock` + `game.clock-offset` + 런타임 이동량). 남은 시간을 다시 재며 최대 [MAX_SLEEP] 씩 기다리고,

    31	 * - 기동 때 08:00 이 이미 지난 가장 늦은 날짜([catchUpDate])가 있으면 그날 1차는 곧바로, 2차는 `max(지금, 08:00:03)` 에 다시 돈다 —

    35	@Component
    36	class DeadlineScheduler(
    37	    private val service: DeadlineService,
    38	    private val clock: GameClock,
    39	    private val game: GameProperties,
    40	    private val properties: DeadlineProperties,
    41	) : SmartLifecycle {
    42	    @Volatile private var job: Job? = null
    43	
    44	    override fun start() {
    45	        if (!properties.enabled || job != null) return
    46	        val scope = CoroutineScope(SupervisorJob() + Dispatchers.Default + CoroutineName("deadline"))
    47	        scope.launch { schedule(scope) }
    48	        job = scope.coroutineContext[Job]
    49	        log.info(
    50	            "deadline.started buckets={} instance={}/{} first={} last={}",
    51	            service.buckets.size, properties.instanceIndex, properties.instanceCount, service.buckets.firstOrNull(), service.buckets.lastOrNull(),
    52	        )
    53	    }
    54	
    55	    override fun stop() {
    56	        val running = job ?: return
    57	        // 도중에 끊긴 단계는 다음 기동의 다시 돌기(멱등)가 마저 한다
    58	        runBlocking { running.cancelAndJoin() }
    59	        job = null
    60	        log.info("deadline.stopped")
    61	    }
    62	
    63	    override fun isRunning(): Boolean = job != null
    64	
    65	    /** 기동 때 다시 돌 날짜 — 08:00 이 지난 가장 늦은 날짜 `d` 이고 `now < d+1 의 07:00` 일 때만(LLD §3.2 7). */
    66	    fun catchUpDate(now: Instant): LocalDate? {
    67	        val today = clock.playDate(now)
    68	        val date = if (clock.deadlineAt(today) <= now) today else today.minusDays(1)
    69	        return date.takeIf { now < clock.startAt(it.plusDays(1)) }
    70	    }
    71	
    72	    /** 08:00 이 아직 오지 않은 가장 이른 날짜. */
    73	    fun nextDate(now: Instant): LocalDate {
    74	        val today = clock.playDate(now)
    75	        return if (now < clock.deadlineAt(today)) today else today.plusDays(1)
    76	    }
    77	
    78	    private suspend fun schedule(scope: CoroutineScope) {
    79	        val now = clock.now()
    80	        val catchUp = catchUpDate(now)
    81	        var date = if (catchUp != null) {
    82	            log.info("deadline.catch_up date={}", catchUp)
    83	            runDay(scope, catchUp)
    84	            catchUp.plusDays(1)
    85	        } else {
    86	            nextDate(now)
    87	        }
    88	        while (true) {
    89	            runDay(scope, date)
    90	            date = date.plusDays(1)
    91	        }
    92	    }
    93	
    94	    /** [date] 의 08:00:00 까지 기다려 1차, 08:00:03 까지 기다려 2차를 띄운다(이미 지났으면 곧바로). */
    95	    private suspend fun runDay(scope: CoroutineScope, date: LocalDate) {
    96	        sleepUntil(clock.deadlineAt(date))
    97	        launchPhase(scope, date, DeadlinePhase.FIRST)
    98	        sleepUntil(game.graceEndAt(date))
    99	        launchPhase(scope, date, DeadlinePhase.SECOND)
   100	    }
   101	
   102	    /** 시각 이동(데모 시계 — LLD platform §4.4-4)에도 깨어 남은 시간을 다시 잰다. */
   103	    private suspend fun sleepUntil(at: Instant) {
   104	        while (!clock.pauseUntil(at, MAX_SLEEP)) Unit
   105	    }
   106	
   107	    private fun launchPhase(scope: CoroutineScope, date: LocalDate, phase: DeadlinePhase) {
   108	        scope.launch(CoroutineName("deadline-$date-${phase.value}")) {
   109	            var backoff = properties.phaseRetryBackoff
   110	            while (true) {
   111	                val clean = try {
   112	                    service.runPhase(date, phase).also(::report).let { it.failedGroups.isEmpty() && it.failedBuckets.isEmpty() }
   113	                } catch (e: CancellationException) {
   114	                    throw e
   115	                } catch (e: Exception) {
   116	                    log.error("deadline.phase_error date={} phase={}", date, phase.value, e)
   117	                    false
   118	                }
   119	                // 1차 실패는 08:00:03 의 2차가 덮는다. 2차 실패는 다음 날 07:00 전까지 다시 돈다(LLD §3.2 8 · 멱등)
   120	                if (clean || phase != DeadlinePhase.SECOND) return@launch
   121	                if (clock.now().plus(backoff) >= clock.startAt(date.plusDays(1))) {
   122	                    log.error("deadline.phase_gave_up date={} phase={}", date, phase.value)
   123	                    return@launch
   124	                }
   125	                log.warn("deadline.phase_retry date={} phase={} afterMs={}", date, phase.value, backoff.toMillis())
   126	                delay(backoff.toMillis())
   127	                backoff = minOf(backoff.multipliedBy(2), properties.phaseRetryBackoffMax)
   128	            }
   129	        }
   130	    }
   131	
   132	    private fun report(r: DeadlineReport) {
   133	        r.failedGroups.forEach {
   134	            log.warn("deadline.group_failed gid={} date={} phase={} reason={}", it.gid.value, r.date, r.phase.value, it.reason)
   135	        }
   136	        r.failedBuckets.forEach { log.warn("deadline.bucket_failed bucket={} date={} phase={}", it, r.date, r.phase.value) }
   137	        log.info(
   138	            "deadline.phase_done date={} phase={} buckets={} groups={} pending={} decided={} decidedFinal={} finalized={} " +
   139	                "skipped={} noGroup={} failed={} failedBuckets={} elapsedMs={}",
   140	            r.date, r.phase.value, r.buckets, r.groups,
   141	            r.count(DeadlineCode.PENDING), r.count(DeadlineCode.DECIDED), r.count(DeadlineCode.DECIDED_FINAL),
   142	            r.count(DeadlineCode.FINALIZED), r.count(DeadlineCode.SKIPPED), r.count(DeadlineCode.NO_GROUP),
   143	            r.failedGroups.size, r.failedBuckets.size, r.elapsed.toMillis(),
   144	        )
   145	    }

    43	 * - 루프는 [GameClock.now](시각 주입 포함) 다음에 오는 07:00 까지 기다렸다가 돈다. 기동 때는 게임 시각의 오늘 · 내일 파티션을 만들고,
    44	 *   07:00 ~ 08:00 사이에 떴으면 오늘 실행을 바로 한다(07:00 실행 중 죽은 경우의 따라잡기 — LLD §3.5 v5.1).
    45	 * - 루프의 실행은 끝까지 성공(건너뛴 페이지 0)할 때까지 그날 08:00 전까지 통째로 다시 돈다([confirmUntilDeadline]) — 멱등이라 다시 돌아도 같다.
    46	 * - 재적재(LLD platform §4.2): 명단을 만들 때 그날 PostgreSQL 기록이 있으면 그것으로 그룹 Hash 를 되살리고, 08:00 이 지났으면 마감을 따라잡는다.
    47	 *   끝까지 성공하면 버킷 표시를 쓰고, 다음 07:00 을 기다리는 동안 [MemberProperties.watchInterval] 마다 [watchOnce] 로 표시가 사라졌는지 본다.
    48	 */

    61	     * [rehydrated] = 그중 PostgreSQL 기록으로 되살린 그룹 수, [caughtUp] = 마감 따라잡기에서 `SKIPPED` 가 아닌 결과를 낸 그룹 수,

    99	    override fun start() {

   102	        scope.launch { loop() }

   115	    private suspend fun loop() {
   116	        val now = gameClock.now()
   117	        val today = gameClock.playDate(now)
   118	        guarded("startup-partitions", today) {
   119	            retrying("partitions") {
   120	                dailyProgress.ensurePartitions(today)
   121	                dailyProgress.ensurePartitions(today.plusDays(1))
   122	            }
   123	        }
   124	        catchUpDate(now)?.let { confirmUntilDeadline(it) }
   125	        while (true) {
   126	            val date = nextRunDate(gameClock.now())
   127	            val at = gameClock.startAt(date)
   128	            // delay 는 단조 시계라 벽시계(시각 주입 포함)로 다시 보고, 아직이면 더 기다린다.
   129	            // 기다리는 동안 지금 게임 날짜(date - 1)의 표시를 감시한다 — 사라지면 Redis 를 잃은 것이라 다시 올린다(LLD platform §4.2)
   130	            val watched = date.minusDays(1)
   131	            // 데모 시각 이동이 있으면 바로 깨어 남은 시간을 다시 잰다(LLD platform §4.4-4)
   132	            while (!gameClock.pauseUntil(at, properties.watchInterval)) {
   133	                guarded("watch", watched) { watchOnce(watched) }
   134	            }
   135	            confirmUntilDeadline(date)
   136	        }
   137	    }

   139	    /**
   140	     * [date] 실행을 건너뛴 페이지 없이 끝날 때까지 그날 08:00 전까지 다시 돈다(사이 [MemberProperties.retryBackoff] 부터 두 배 · 최대 [MAX_BACKOFF]).
   141	     * 08:00 이 지나면 그만둔다 — 마감이 그날 그룹 목록을 이미 읽었다. 그동안 요청은 §3.4 로 명단을 그 자리에서 만든다.
   142	     */
   143	    private suspend fun confirmUntilDeadline(date: LocalDate) {
   144	        var backoff = properties.retryBackoff
   145	        while (true) {
   146	            var done = false
   147	            guarded("run", date) { done = run(date).complete }
   148	            if (done) return
   149	            if (gameClock.now().plus(backoff) >= gameClock.deadlineAt(date)) {
   150	                log.error("member.gave-up date={} — 08:00 전에 끝내지 못했다", date)
   151	                return
   152	            }
   153	            delay(backoff.toMillis())
   154	            backoff = minOf(backoff.multipliedBy(2), MAX_BACKOFF)
   155	        }
   156	    }
   157	
   158	    /**
   159	     * 감시 한 번(LLD platform §4.2 발동 2) — [date] 의 버킷 표시를 센다. 다 있으면 그 날짜를 「본 날짜」로 두고 `false`.
   160	     * 본 적 있는 날짜의 표시가 하나라도 없으면 Redis 를 잃은 것으로 보고 [run] 을 다시 돈 뒤 `true`(08:00 이 지났어도 — 마감은 따라잡는다).
   161	     * 앞 실행이 끝까지 되지 않았으면([Summary.complete] 거짓 — 따라잡기 실패 등) 표시가 다 있어도 다시 돈다.
   162	     * 본 적 없는 날짜면 돌지 않는다(`false`) — 08:00 뒤 기동 때 `run` 을 하지 않는 규칙(game-session §3.5)을 지킨다.
   163	     * [startAfter] 는 [run] 에 그대로 넘긴다(테스트 — 루프는 늘 처음부터).
   164	     */
   165	    suspend fun watchOnce(date: LocalDate, startAfter: GroupId? = null): Boolean {
   166	        val missing = rosters.countUnmarked(date)
   167	        if (missing > 0 && markedDate != date) return false
   168	        if (missing == 0 && retryDate != date) {
   169	            markedDate = date
   170	            return false
   171	        }
   172	        if (missing > 0) {
   173	            log.warn("member.redis_lost date={} missing={} — 그날 명단 · 그룹 Hash 를 다시 올린다", date, missing)
   174	        } else {
   175	            log.warn("member.rerun date={} — 앞 실행이 끝까지 되지 않았다", date)
   176	        }
   177	        run(date, startAfter)
   178	        return true
   179	    }

   181	    /** 다음 실행 날짜 — [now] 보다 뒤인 가장 이른 07:00 의 날짜. */
   182	    internal fun nextRunDate(now: Instant): LocalDate {
   183	        val today = gameClock.playDate(now)
   184	        return if (now < gameClock.startAt(today)) today else today.plusDays(1)
   185	    }
   186	
   187	    /** 기동 때 바로 돌 날짜 — 오늘 07:00 ~ 08:00 이면 오늘, 아니면 없음(08:00 뒤에는 마감이 이미 그날 그룹 목록을 읽었다). */
   188	    internal fun catchUpDate(now: Instant): LocalDate? {
   189	        val today = gameClock.playDate(now)
   190	        return today.takeIf { now >= gameClock.startAt(it) && now < gameClock.deadlineAt(it) }
   191	    }

   193	    /**
   194	     * [date] 의 명단 확정(LLD §3.5). 기다리지 않고 바로 돌고, 시각을 검사하지 않는다 — 루프가 `D 07:00` 뒤에만 부른다.
   195	     * 파티션 · 페이지 읽기가 끝내 실패하면 예외로 끝나고, 한 페이지 처리가 끝내 실패하면 그 페이지를 건너뛴다.
   196	     * [startAfter] 는 그 그룹 다음부터 돈다(테스트가 잡을 특정 그룹에 바로 닿게 할 때 — 루프는 늘 처음부터).
   197	     */
   198	    suspend fun run(date: LocalDate, startAfter: GroupId? = null): Summary {
   199	        val started = System.nanoTime()
   200	        val cutoff = gameClock.joinCutoffAt(date)
   201	        retrying("partitions") {
   202	            dailyProgress.ensurePartitions(date)
   203	            dailyProgress.ensurePartitions(date.plusDays(1))
   204	        }
   205	        var total = 0
   206	        var playing = 0
   207	        var solo = 0
   208	        var empty = 0
   209	        var created = 0
   210	        var rehydrated = 0
   211	        var caughtUp = 0
   212	        var failedPages = 0
   213	        var failedGroups = 0
   214	        // 표시는 시작할 때 쓴다 — 실행 도중 Redis 가 비면 끝에서 다시 셀 때 빠져 있어 실패로 본다(LLD platform §4.2)
   215	        retrying("mark") { rosters.markConfirmed(date) }
   216	        markedDate = date
   217	        var after: GroupId? = startAfter
   218	        while (true) {
   219	            val page = retrying("groups") { groups.findPage(after, properties.pageSize) }
   220	            if (page.isEmpty()) break
   221	            total += page.size
   222	            try {
   223	                // 다시 해도 같다 — roster 는 없을 때만, Q10 은 ON CONFLICT DO NOTHING
   224	                val r = retrying("page") { confirmPage(page, date, cutoff) }
   225	                playing += r.playing
   226	                solo += r.solo
   227	                empty += r.empty
   228	                created += r.created
   229	                rehydrated += r.rehydrated
   230	                caughtUp += r.caughtUp
   231	                failedGroups += r.failedGroups
   232	            } catch (e: CancellationException) {
   233	                throw e
   234	            } catch (e: Exception) {
   235	                // 이 페이지 그룹은 요청 때 RosterService 가 명단을 만든다(§3.4)
   236	                failedPages++
   237	                log.warn("member.page-skipped date={} from={} size={}", date, page.first().first.value, page.size, e)
   238	            }
   239	            if (page.size < properties.pageSize) break
   240	            after = page.last().first
   241	        }
   242	        val lost = retrying("mark-check") { rosters.countUnmarked(date) }
   243	        if (lost > 0) {
   244	            log.warn("member.redis_lost date={} missing={} — 실행 도중 표시가 사라졌다", date, lost)
   245	            failedGroups++
   246	        }
   247	        val summary = Summary(
   248	            date, total, playing, solo, empty, created, failedPages, Duration.ofNanos(System.nanoTime() - started), rehydrated, caughtUp,
   249	            failedGroups,
   250	        )
   251	        retryDate = if (summary.complete) null else date
   252	        log.info(
   253	            "member.done date={} groups={} playing={} solo={} empty={} created={} rehydrated={} caughtUp={} failedPages={} " +
   254	                "failedGroups={} elapsedMs={}",
   255	            date, summary.groups, summary.playing, summary.solo, summary.empty, summary.created, summary.rehydrated, summary.caughtUp,
   256	            summary.failedPages, summary.failedGroups, summary.elapsed.toMillis(),
   257	        )
   258	        return summary
   259	    }

   261	    /**
   262	     * 한 페이지: Q4 한 번 → 2명 이상 그룹의 그날 기록 R1 · R2 한 번(LLD platform §4.2) → `roster` 동시(기록이 있으면 재적재 꼬리) → Q10 한 번.
   263	     * 재적재로 만든 그룹은 지금 ≥ 08:00 이면 마감을 따라잡는다.
   264	     */
   265	    private suspend fun confirmPage(page: List<Pair<GroupId, GroupName>>, date: LocalDate, cutoff: Instant): PageResult {
   266	        val members = users.findMembersJoinedBefore(page.map { it.first }, cutoff)
   267	        val built = page.map { (gid, name) -> DayRoster(gid, date, name, members[gid].orEmpty()) }
   268	        val playingGids = built.filter { it.size >= MIN_PLAYING }.map { it.gid }
   269	        val records = dailyProgress.loadDayStates(date, playingGids)
   270	        val now = gameClock.now()
   271	        val deadlineAt = gameClock.deadlineAt(date)
   272	        // 기록이 규칙 밖인 그룹은 그 그룹만 건너뛴다 — 페이지의 다른 그룹까지 막지 않는다(실패로 세어 다시 돈다)
   273	        val bad = mutableSetOf<GroupId>()
   274	        val restores = built.map { r ->
   275	            if (r.size < MIN_PLAYING) return@map null
   276	            try {
   277	                GroupRestore.of(r, records[r.gid], now, deadlineAt)
   278	            } catch (e: IllegalArgumentException) {
   279	                log.error("member.rehydrate_bad gid={} date={}", r.gid.value, date, e)
   280	                bad += r.gid
   281	                null
   282	            }
   283	        }
   284	        val permits = Semaphore(properties.concurrency)
   285	        val made = coroutineScope {
   286	            built.indices.map { i ->
   287	                async { if (built[i].gid in bad) false else permits.withPermit { rosters.createIfAbsent(built[i], restores[i]) } }
   288	            }.awaitAll()
   289	        }
   290	        built.indices.filter { made[it] && restores[it] != null }.forEach { log.warn("member.rehydrated gid={} date={}", built[it].gid.value, date) }
   291	        // 마감 따라잡기는 기록이 있는 그룹 모두 — 이번에 만들었는지와 상관없다(페이지 재시도 · 앞 실행의 실패에도 빠지지 않게, Lua 는 멱등)
   292	        val restored = built.indices.filter { restores[it] != null }.map { built[it].gid }
   293	        val outcomes = if (restored.isNotEmpty() && gameClock.now() >= deadlineAt) {
   294	            coroutineScope { restored.map { gid -> async { permits.withPermit { catchUpDeadline(gid, date) } } }.awaitAll() }
   295	        } else {
   296	            emptyList()
   297	        }
   298	        // created 와 상관없이 넣는다 — api 가 먼저 만든 명단의 -1 행이 빠졌어도 채운다(§3.5-4)
   299	        val rows = built.filter { it.size >= MIN_PLAYING }.flatMap { r -> r.members.keys.map { it to r.gid } }
   300	        dailyProgress.insertNotStarted(date, rows)
   301	        return PageResult(
   302	            playing = built.count { it.size >= MIN_PLAYING },
   303	            solo = built.count { it.size == 1 },
   304	            empty = built.count { it.size == 0 },
   305	            created = made.count { it },
   306	            rehydrated = built.indices.count { made[it] && restores[it] != null },
   307	            caughtUp = outcomes.count { it != null && it != DeadlineCode.SKIPPED },
   308	            failedGroups = bad.size + outcomes.count { it == null },
   309	        )
   310	    }
   311	
   312	    /**
   313	     * 마감 따라잡기(LLD platform §4.2 발동 3) — 08:00 뒤 되살린 그룹은 마감 잡이 이미 지나갔으므로 Lua `deadline` 1차를 한 번 부른다
   314	     * (08:00:03 뒤면 Lua 가 확정까지 한다). 결과 코드를 돌려주고, 끝내 실패하면 null — 실행을 실패로 세어 루프 · 감시가 다시 부른다.
   315	     */
   316	    private suspend fun catchUpDeadline(gid: GroupId, date: LocalDate): DeadlineCode? {
   317	        try {
   318	            repeat(EARLY_RETRIES) {
   319	                val outcome = retrying("deadline") { groupDays.runDeadline(gid, date, DeadlinePhase.FIRST) }
   320	                if (outcome.code != DeadlineCode.EARLY) {
   321	                    log.info("member.caught_up gid={} date={} code={} version={}", gid.value, date, outcome.code, outcome.version)
   322	                    return outcome.code
   323	                }
   324	                // 워커 시계가 Redis 보다 빠름 — deadline-result §3.2 의 EARLY 와 같이 조금 뒤 다시
   325	                delay(EARLY_DELAY_MS)
   326	            }
   327	            log.error("member.catch_up_failed gid={} date={} reason=EARLY", gid.value, date)
   328	        } catch (e: CancellationException) {
   329	            throw e
   330	        } catch (e: Exception) {
   331	            log.error("member.catch_up_failed gid={} date={}", gid.value, date, e)
   332	        }
   333	        return null
   334	    }
   335	
   336	    /** 루프의 한 단계 — 실패는 로그만 남기고 다음 07:00 을 기다린다(그동안 요청은 §3.4 로 명단을 만든다). */
   337	    private suspend fun guarded(what: String, date: LocalDate, block: suspend () -> Unit) {
   338	        try {
   339	            block()
   340	        } catch (e: CancellationException) {
   341	            throw e
   342	        } catch (e: Exception) {
   343	            log.error("member.failed step={} date={}", what, date, e)
   344	        }
   345	    }
   346	
   347	    /** [RETRIES] 번까지 — 사이에 [MemberProperties.retryBackoff] 부터 두 배씩 쉰다. */
   348	    private suspend fun <T> retrying(what: String, block: suspend () -> T): T {
   349	        var backoff = properties.retryBackoff
   350	        repeat(RETRIES - 1) { attempt ->
   351	            try {
   352	                return block()
   353	            } catch (e: CancellationException) {
   354	                throw e
   355	            } catch (e: Exception) {
   356	                log.warn("member.retry what={} attempt={} cause={}", what, attempt + 1, e.toString())
   357	            }
   358	            delay(backoff.toMillis())
   359	            backoff = backoff.multipliedBy(2)
   360	        }
   361	        return block()
   362	    }
   363	
   364	    private companion object {
   365	        const val RETRIES = 3
   366	        const val EARLY_RETRIES = 200
   367	        const val EARLY_DELAY_MS = 20L
   368	        val MAX_BACKOFF: Duration = Duration.ofSeconds(30)
   369	
   370	        /** 그룹 Hash · -1 행을 두는 최소 인원(1명 그룹은 결과가 없다 — HLD §5). `roster.lua` 의 n ≥ 2 와 같다. */
   371	        const val MIN_PLAYING = 2
   372	        val log = LoggerFactory.getLogger(MemberJob::class.java)
   373	    }

    41	    private val users: UserRepository,
    42	    private val questions: QuestionRepository,
    43	    private val banks: QuestionBankCache,
    44	    private val gameClock: GameClock,

    48	    /** 한 번 실행의 요약(LLD §3.6-6). [existing] = 이미 키가 있던 유저(06:30 실행 · `api` 가 만든 것). */
    49	    data class Summary(
    50	        val date: LocalDate,
    51	        val index: Int,
    52	        val targets: Int,
    53	        val created: Int,
    54	        val existing: Int,
    55	        val exhausted: Int,
    56	        val failedBatches: Int,
    57	        val elapsed: Duration,
    58	    ) {
    59	        val usersPerSecond: Long get() = targets * 1_000L / maxOf(1L, elapsed.toMillis())
    60	    }

    62	    /** 실행 하나 — [date] 의 [index] 번째 prepare 시각 [at]. */
    63	    data class Run(val date: LocalDate, val index: Int, val at: Instant)
    64	
    65	    @Volatile private var job: Job? = null
    66	
    67	    override fun start() {
    68	        if (!properties.enabled || job != null) return
    69	        val scope = CoroutineScope(SupervisorJob() + Dispatchers.Default + CoroutineName("prepare-questions"))
    70	        scope.launch { loop() }
    71	        job = scope.coroutineContext[Job]
    72	        log.info("prepare-questions.started times={}", game.prepareTimes)

    83	    private suspend fun loop() {
    84	        var last: Run? = null
    85	        while (true) {
    86	            val next = nextRun(gameClock.now(), last)
    87	            // delay 는 단조 시계라 벽시계(시각 주입 포함)로 다시 보고, 아직이면 더 기다린다.
    88	            // 데모 시각 이동이 있으면 바로 깨어 다시 잰다 — 지난 실행은 곧바로 돈다(LLD platform §4.4-4)
    89	            while (!gameClock.pauseUntil(next.at, MAX_PAUSE)) Unit
    90	            try {
    91	                run(next.date, next.index)
    92	            } catch (e: CancellationException) {
    93	                throw e
    94	            } catch (e: Exception) {
    95	                // 이번 실행은 버린다 — 은행이 없는 유저는 api 가 요청 때 만든다(D18)
    96	                log.error("prepare-questions.failed date={} index={}", next.date, next.index, e)
    97	            }
    98	            last = next
    99	        }
   100	    }
   101	
   102	    /**
   103	     * 다음 실행. [last](방금 끝난 실행) 다음 순번이 같은 날에 있고 그날 마감 전이면 그 시각이 지났어도 그것이다 —
   104	     * 06:30 실행이 06:50 을 넘겨 끝나도 06:50 실행이 빠지지 않는다(LLD §3.6). 아니면 [now] 보다 뒤인 가장 이른 시각.
   105	     */
   106	    internal fun nextRun(now: Instant, last: Run? = null): Run {
   107	        if (last != null && last.index + 1 < game.prepareTimes.size && now < gameClock.deadlineAt(last.date)) {
   108	            return Run(last.date, last.index + 1, gameClock.prepareAt(last.date)[last.index + 1])
   109	        }
   110	        val today = gameClock.playDate(now)
   111	        for (date in listOf(today, today.plusDays(1))) {
   112	            gameClock.prepareAt(date).forEachIndexed { i, at -> if (at > now) return Run(date, i, at) }
   113	        }
   114	        error("다음 prepare 시각이 없다: ${game.prepareTimes}")
   115	    }
   116	
   117	    /** [date] 의 [index] 번째 실행(LLD §3.6). 기다리지 않고 바로 돈다. */
   118	    suspend fun run(date: LocalDate, index: Int): Summary {
   119	        val times = gameClock.prepareAt(date)
   120	        require(index in times.indices) { "prepare 순번은 0 ~ ${times.size - 1} 이다: $index" }
   121	        val from = if (index == 0) null else times[index - 1]
   122	        val to = times[index]
   123	        val expireAt = gameClock.deadlineAt(date)
   124	        val started = System.nanoTime()
   125	        // 잡마다 한 번 — 새로 넣은 문제는 다음 실행부터 들어간다(LLD §5)
   126	        val ids = retrying("questions") { questions.loadIdsByDifficulty() }
   127	        var targets = 0
   128	        var created = 0
   129	        var exhausted = 0
   130	        var failedBatches = 0
   131	        var failedUsers = 0
   132	        var after: UserId? = null
   133	        while (true) {
   134	            // 대상 읽기가 끝내 실패하면 다음 페이지를 알 수 없어 실행을 멈춘다(예외가 루프로 간다)
   135	            val page = retrying("users") { users.findIdsJoinedIn(from, to, after, properties.batchSize) }
   136	            if (page.isEmpty()) break
   137	            targets += page.size
   138	            try {
   139	                // 다시 해도 같다 — NX 라 앞 시도에서 둔 키는 「이미 있음」으로 세일 뿐이다
   140	                val (c, e) = retrying("batch") { prepareBatch(page, ids, date, expireAt) }
   141	                created += c
   142	                exhausted += e
   143	            } catch (e: CancellationException) {
   144	                throw e
   145	            } catch (e: Exception) {
   146	                failedBatches++
   147	                failedUsers += page.size
   148	                log.warn("prepare-questions.batch-skipped date={} index={} from={} size={}", date, index, page.first().value, page.size, e)
   149	            }
   150	            if (page.size < properties.batchSize) break
   151	            after = page.last()
   152	        }
   153	        val summary = Summary(
   154	            date = date,
   155	            index = index,
   156	            targets = targets,
   157	            created = created,
   158	            existing = targets - exhausted - created - failedUsers,
   159	            exhausted = exhausted,
   160	            failedBatches = failedBatches,
   161	            elapsed = Duration.ofNanos(System.nanoTime() - started),
   162	        )
   163	        log.info(
   164	            "prepare-questions.done date={} at={} targets={} created={} existing={} exhausted={} failedBatches={} elapsedMs={} usersPerSec={}",
   165	            date, game.prepareTimes[index], summary.targets, summary.created, summary.existing, summary.exhausted,
   166	            summary.failedBatches, summary.elapsed.toMillis(), summary.usersPerSecond,
   167	        )
   168	        return summary
   169	    }
   170	
   171	    /** 한 묶음: 출제 이력(Q3) → 뽑기 → `SET NX PXAT` 파이프라인. (새로 둔 수, 고갈 수). */
   172	    private suspend fun prepareBatch(
   173	        page: List<UserId>,
   174	        ids: Map<Difficulty, LongArray>,
   175	        date: LocalDate,
   176	        expireAt: Instant,
   177	    ): Pair<Int, Int> {
   178	        val served = questions.findServedIds(page)
   179	        val built = LinkedHashMap<UserId, QuestionBank>(page.size * 2)
   180	        var exhausted = 0
   181	        for (uid in page) {
   182	            when (val draw = QuestionBankBuilder.build(ids, served[uid].orEmpty(), game.bankSizePerDifficulty)) {
   183	                is BankDraw.Built -> built[uid] = draw.bank
   184	                is BankDraw.Exhausted -> exhausted++
   185	            }
   186	        }
   187	        return banks.putAllIfAbsent(date, built, expireAt) to exhausted
   188	    }
   189	
   190	    /** [RETRIES] 번까지 — 사이에 [PrepareQuestionsProperties.retryBackoff] 부터 두 배씩 쉰다. */
   191	    private suspend fun <T> retrying(what: String, block: suspend () -> T): T {
   192	        var backoff = properties.retryBackoff
   193	        repeat(RETRIES - 1) { attempt ->
   194	            try {
   195	                return block()
   196	            } catch (e: CancellationException) {
   197	                throw e
   198	            } catch (e: Exception) {
   199	                log.warn("prepare-questions.retry what={} attempt={} cause={}", what, attempt + 1, e.toString())
   200	            }
   201	            delay(backoff.toMillis())
   202	            backoff = backoff.multipliedBy(2)
   203	        }
   204	        return block()
   205	    }
   206	
   207	    private companion object {
   208	        const val RETRIES = 3
   209	        val log = LoggerFactory.getLogger(PrepareQuestionsJob::class.java)
   210	
   211	        /** 한 번에 기다리는 상한 — 벽시계로 다시 잰다(시각 이동은 상한과 무관하게 바로 깨운다). */
   212	        val MAX_PAUSE: Duration = Duration.ofMinutes(1)
   213	    }
   214	}
