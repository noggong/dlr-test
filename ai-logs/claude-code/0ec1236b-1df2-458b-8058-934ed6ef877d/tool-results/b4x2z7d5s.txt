    10	    /** 프리셋이 경계 앞에 서는 간격 — `start` = 07:00 − 10초 · `deadline` = 08:00 − 10초. */
    11	    val PRESET_LEAD: Duration = Duration.ofSeconds(10)
    12	
    13	    enum class Preset { START, DEADLINE }
    14	
    15	    /** 이동 검사 결과 — 통과면 [ok]. 거절이면 [reason] 이 이유. */
    16	    data class Check(val ok: Boolean, val reason: String)
    17	
    18	    /**
    19	     * 첫 런타임 이동량(P10 · LLD §4.4-1): 실제 [realNow] + [lead] 가 게임 `D 07:00` 이 되게 한다.
    20	     * `D` 는 [realNow] 의 KST 날짜부터, 이동 총량(`clock-offset` + 결과)이 `start-time − join-cutoff` 이상이고 결과가 0 이상인 가장 이른 날짜다.
    21	     */
    22	    fun first(realNow: Instant, lead: Duration, game: GameProperties): Duration {
    23	        val gap = Duration.between(game.joinCutoff, game.startTime)
    24	        val target = realNow.plus(lead)
    25	        var date = realNow.atZone(game.zone).toLocalDate()
    26	        while (true) {
    27	            val total = Duration.between(target, game.startAt(date))
    28	            val shift = total.minus(game.clockOffset)
    29	            if (total >= gap && !shift.isNegative) return shift
    30	            date = date.plusDays(1)
    31	        }
    32	    }
    33	
    34	    /** [now] 뒤(>) 가장 이른 경계 — 그날 · 다음 날의 07:00 명단 확정 · 08:00 마감 · 08:00:03 랭킹 확정. */
    35	    fun nextBoundary(now: Instant, game: GameProperties): Instant {
    36	        val today = now.atZone(game.zone).toLocalDate()
    37	        return (-1L..2L).asSequence()
    38	            .map { today.plusDays(it) }
    39	            .flatMap { boundaries(it, game) }
    40	            .filter { it > now }
    41	            .min()
    42	    }
    43	
    44	    /** 지금 옮길 수 있는 가장 늦은 시각 = 다음 경계 − [margin]. */
    45	    fun moveLimit(now: Instant, margin: Duration, game: GameProperties): Instant = nextBoundary(now, game).minus(margin)
    46	
    47	    /** 프리셋 시각 — [now] 뒤 가장 이른 07:00(`START`) · 08:00(`DEADLINE`)의 [PRESET_LEAD] 전. */
    48	    fun preset(now: Instant, preset: Preset, game: GameProperties): Instant {
    49	        val today = now.atZone(game.zone).toLocalDate()
    50	        val at = { d: LocalDate -> if (preset == Preset.START) game.startAt(d) else game.deadlineAt(d) }
    51	        val next = generateSequence(today) { it.plusDays(1) }.map(at).first { it > now }
    52	        return next.minus(PRESET_LEAD)
    53	    }
    54	
    55	    /** P12 — `now < target ≤ 다음 경계 − margin` 이면 통과. */
    56	    fun check(now: Instant, target: Instant, margin: Duration, game: GameProperties): Check {
    57	        if (target <= now) return Check(false, "앞으로만 옮긴다: now=$now target=$target")

    59	        if (target > limit) return Check(false, "다음 경계 ${nextBoundary(now, game)} 의 ${margin.seconds}초 전($limit)까지만 옮긴다: target=$target")

    61	    }

    96	        repeat(MOVE_ATTEMPTS) {
    97	            val seen = store.get() ?: game.shift.micros
    98	            apply(seen)
    99	            val now = game.now(clock)
   100	            val at = when (target) {
   101	                is Target.Preset -> DemoClockRules.preset(now, target.preset, game)
   102	                is Target.At -> target.at
   103	            }

   105	            if (!check.ok) throw BusinessException(ErrorCode.CLOCK_MOVE_REJECTED, "${ErrorCode.CLOCK_MOVE_REJECTED.defaultMessage} ${check.reason}")

   108	                apply(next)
   109	                return snapshot()
   110	            }
   111	        }

   118	    }

   122	        if (from == value) return

   125	    }
